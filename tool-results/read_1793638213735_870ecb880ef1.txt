    72→/** Rounded canvas every figure draws on (default 360×200). */
    73→function Frame({
    74→  label,
    75→  children,
    76→  height = 200,
    77→}: {
    78→  label: string

    80→  height?: number

    94→/** Teal / muted / rose arrowhead marker — must live inside <defs> with a unique id. */
    95→function ArrowHead({ id, tone = "teal" }: { id: string; tone?: "teal" | "muted" | "rose" }) {
    96→  const fill =
    97→    tone === "muted"
    98→      ? "fill-muted-foreground/60"
    99→      : tone === "rose"
   100→        ? "fill-rose-500"
   101→        : "fill-teal-500"
   102→  return (
   103→    <marker
   104→      id={id}
   105→      viewBox="0 0 8 8"
   106→      refX={7}
   107→      refY={4}
   108→      markerWidth={6.5}
   109→      markerHeight={6.5}
   110→      orient="auto-start-reverse"
   111→    >
   112→      <path d="M0 0L8 4L0 8Z" className={fill} />
   113→    </marker>
   114→  )
   115→}

   117→/** A UI node box with optional centered label. */
   118→function Box({
   119→  x,
   120→  y,
   121→  w,
   122→  h,
   123→  label,
   124→  rx = 6,
   125→  className,
   126→}: {
   127→  x: number
   128→  y: number
   129→  w: number
   130→  h: number
   131→  label?: string
   132→  rx?: number
   133→  className?: string
   134→}) {
   135→  return (
   136→    <g>
   137→      <rect x={x} y={y} width={w} height={h} rx={rx} strokeWidth={1.5} className={cn(panelCls, className)} />
   138→      {label !== undefined && (
   139→        <text x={x + w / 2} y={y + h / 2 + 3} fontSize={8.5} textAnchor="middle" className="fill-foreground/80">
   140→          {label}
   141→        </text>
   142→      )}
   143→    </g>
   144→  )
   145→}

   147→/** Teal selection grips on the four corners of a box. */
   148→function Grips({ x, y, w, h }: { x: number; y: number; w: number; h: number }) {
   149→  const s = 5
   150→  const pts: Array<[number, number]> = [
   151→    [x - s / 2, y - s / 2],
   152→    [x + w - s / 2, y - s / 2],
   153→    [x - s / 2, y + h - s / 2],
   154→    [x + w - s / 2, y + h - s / 2],
   155→  ]
   156→  return (
   157→    <g>
   158→      {pts.map(([gx, gy], i) => (
   159→        <rect
   160→          key={i}
   161→          x={gx}
   162→          y={gy}
   163→          width={s}
   164→          height={s}
   165→          rx={1}
   166→          strokeWidth={1.2}
   167→          className="fill-background stroke-teal-500"
   168→        />
   169→      ))}
   170→    </g>
   171→  )
   172→}

   209→/**
   210→ * A rounded UI button tile with the REAL lucide icon inside (nested svg),
   211→ * exactly the glyph the board renders for that control. `active` reproduces
   212→ * the teal highlight the UI uses for the selected tool / panel.
   213→ */
   214→function Tile({
   215→  x,
   216→  y,
   217→  size = 14,
   218→  icon,
   219→  active = false,
   220→  tileClassName,
   221→  iconClassName,
   222→}: {
   223→  x: number
   224→  y: number
   225→  size?: number
   226→  icon: LucideIcon
   227→  active?: boolean
   228→  tileClassName?: string
   229→  iconClassName?: string
   230→}) {
   231→  const iconSize = size * 0.78
   232→  const Icon = icon
   233→  return (
   234→    <g>
   235→      <rect
   236→        x={x}
   237→        y={y}
   238→        width={size}
   239→        height={size}
   240→        rx={size * 0.26}
   241→        strokeWidth={1.2}
   242→        className={cn(
   243→          active ? "fill-teal-500/15 stroke-teal-500" : "fill-card stroke-border",
   244→          tileClassName,
   245→        )}
   246→      />
   247→      <Icon
   248→        x={x + (size - iconSize) / 2}
   249→        y={y + (size - iconSize) / 2}
   250→        width={iconSize}
   251→        height={iconSize}
   252→        aria-hidden
   253→        className={cn(
   254→          active ? "text-teal-600 dark:text-teal-300" : "text-foreground/75",
   255→          iconClassName,
   256→        )}
   257→      />
   258→    </g>
   259→  )
   260→}

   262→/**
   263→ * Bottom-bar pill button showing icon + tiny caption, mirroring the real
   264→ * style-bar buttons ("Shapes", "BG", "Border", …).

   266→function BarBtn({
   267→  x,
   268→  y,
   269→  w = 34,
   270→  h = 18,
   271→  icon,
   272→  label,
   273→  textGlyph,
   274→  active = false,
   275→}: {
   276→  x: number
   277→  y: number
   278→  w?: number
   279→  h?: number
   280→  icon?: LucideIcon
   281→  label?: string
   282→  /** Renders a text glyph (e.g. "T", "Aa") instead of an icon. */
   283→  textGlyph?: string
   284→  active?: boolean
   285→}) {
   286→  const Icon = icon
   287→  return (
   288→    <g>

   295→        strokeWidth={1.1}
   296→        className={cn(active ? "fill-teal-500/15 stroke-teal-500" : "fill-background stroke-border")}
   297→      />
   298→      {Icon && label && (
   299→        <>
   300→          <Icon
   301→            x={x + 3.5}
   302→            y={y + (h - 9) / 2}
   303→            width={9}
   304→            height={9}
   305→            aria-hidden
   306→            className="text-teal-600 dark:text-teal-400"
   307→          />
   308→          <text x={x + 14} y={y + h / 2 + 2.4} fontSize={6.2} className="fill-foreground/80">
   309→            {label}
   310→          </text>
   311→        </>
   312→      )}
   313→      {textGlyph && (
   314→        <text
   315→          x={x + w / 2}
   316→          y={y + h / 2 + 3.2}
   317→          fontSize={9}
   318→          fontWeight={700}
   319→          textAnchor="middle"
   320→          className="fill-foreground/80"
   321→        >
   322→          {textGlyph}
   323→        </text>
   324→      )}
   325→    </g>
   326→  )
   327→}
