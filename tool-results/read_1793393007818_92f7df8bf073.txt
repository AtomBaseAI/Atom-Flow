   103→// Unique id for programmatically created nodes — Date.now() alone collides
   104→// when several elements are created within the same millisecond (e.g. rapid
   105→// double-adds), which corrupts the node lookup.
   106→let nodeSeq = 0
   107→function newNodeId(): string {

   109→  return `n-${Date.now().toString(36)}-${nodeSeq.toString(36)}-${Math.random().toString(36).slice(2, 6)}`

   115→// View-only lock mode (persists across reloads): hides the editing chrome,
   116→// forces the hand tool and freezes node/edge interactions while panning and

   118→const LOCK_STORAGE_KEY = "atomflow:board:locked:v1"

   120→// The lock flag lives in localStorage (an external system), so it is exposed
   121→// through useSyncExternalStore: no setState-in-effect, and the server snapshot
   122→// (unlocked) prevents hydration mismatches while still re-rendering locked
   123→// right after hydration when the stored flag is "1".
   124→const lockListeners = new Set<() => void>()

   126→function subscribeLock(onChange: () => void) {
   127→  lockListeners.add(onChange)
   128→  return () => {
   129→    lockListeners.delete(onChange)
   130→  }
   131→}

   186→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   187→function pickSize(value: unknown): number | undefined {
   188→  const n = typeof value === "string" ? Number(value) : (value as number)
   189→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   190→}
   191→
   192→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   193→// explicit width/height (written by NodeResizer) → measured (actual rendered
   194→// size) → style (size at creation). Resizing never updates node.style, so a
   195→// round trip that kept only style would silently reset resized elements to
   196→// their creation dimensions.
   197→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   198→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   199→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   200→  return { width, height }
   201→}

   203→function sanitizeImportPayload(raw: unknown): {
   204→  nodes: Node<ShapeData>[]
   205→  edges: Edge[]
   206→  viewport?: { x: number; y: number; zoom: number }
   207→} {
   208→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   209→    throw new Error('expected a JSON object with a "nodes" array')
   210→  }
   211→  const obj = raw as Record<string, unknown>
   212→  if (!Array.isArray(obj.nodes)) {
   213→    throw new Error('missing "nodes" array')
   214→  }
   215→  const validTypes = new Set(Object.keys(nodeTypes))
   216→  const stamp = Date.now()
   217→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   218→    const record = (n ?? {}) as Record<string, any>
   219→    // Restore the exact exported size: prefer the resized width/height, then
   220→    // the measured size, then the creation style — and keep style in sync so
   221→    // React Flow (node.width ?? node.style.width) renders the same box.
   222→    const { width, height } = resolveNodeSizeRecord(record)
   223→    return {
   224→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   225→      type: validTypes.has(record.type) ? record.type : "rectangle",
   226→      position: {
   227→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   228→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   229→      },
   230→      data: { ...(record.data ?? {}) },
   231→      style: {
   232→        ...(record.style ?? {}),
   233→        ...(width !== undefined ? { width } : {}),
   234→        ...(height !== undefined ? { height } : {}),
   235→      },
   236→      ...(width !== undefined ? { width } : {}),
   237→      ...(height !== undefined ? { height } : {}),
   238→      // Deliberately no `measured` — it belongs to the exporting session and
   239→      // React Flow re-measures the node on mount.
   240→      selected: false,
   241→      dragging: false,
   242→      resizing: false,
   243→    } as unknown as Node<ShapeData>
   244→  })
   245→  const nodeIds = new Set(nodes.map((n) => n.id))
   246→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   247→    .filter(
   248→      (e) =>
   249→        e &&
   250→        typeof e.source === "string" &&
   251→        typeof e.target === "string" &&
   252→        nodeIds.has(e.source) &&
   253→        nodeIds.has(e.target) &&
   254→        e.source !== e.target,
   255→    )
   256→    .map((e, i) => ({
   257→      ...e,
   258→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   259→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   260→      selected: false,
   261→    })) as Edge[]
   262→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   263→  return { nodes, edges, viewport }
   264→}
