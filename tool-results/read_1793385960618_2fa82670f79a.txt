   114→   114→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   115→   115→function pickSize(value: unknown): number | undefined {
   116→   116→  const n = typeof value === "string" ? Number(value) : (value as number)
   117→   117→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   118→   118→}
   119→   119→
   120→   120→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   121→   121→// explicit width/height (written by NodeResizer) → measured (actual rendered
   122→   122→// size) → style (size at creation). Resizing never updates node.style, so a
   123→   123→// round trip that kept only style would silently reset resized elements to
   124→   124→// their creation dimensions.
   125→   125→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   126→   126→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   127→   127→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   128→   128→  return { width, height }
   129→   129→}

   131→   131→function sanitizeImportPayload(raw: unknown): {
   132→   132→  nodes: Node<ShapeData>[]
   133→   133→  edges: Edge[]
   134→   134→  viewport?: { x: number; y: number; zoom: number }
   135→   135→} {
   136→   136→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   137→   137→    throw new Error('expected a JSON object with a "nodes" array')
   138→   138→  }
   139→   139→  const obj = raw as Record<string, unknown>
   140→   140→  if (!Array.isArray(obj.nodes)) {
   141→   141→    throw new Error('missing "nodes" array')
   142→   142→  }
   143→   143→  const validTypes = new Set(Object.keys(nodeTypes))
   144→   144→  const stamp = Date.now()
   145→   145→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   146→   146→    const record = (n ?? {}) as Record<string, any>
   147→   147→    // Restore the exact exported size: prefer the resized width/height, then
   148→   148→    // the measured size, then the creation style — and keep style in sync so
   149→   149→    // React Flow (node.width ?? node.style.width) renders the same box.
   150→   150→    const { width, height } = resolveNodeSizeRecord(record)
   151→   151→    return {
   152→   152→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   153→   153→      type: validTypes.has(record.type) ? record.type : "rectangle",
   154→   154→      position: {
   155→   155→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   156→   156→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   157→   157→      },
   158→   158→      data: { ...(record.data ?? {}) },
   159→   159→      style: {
   160→   160→        ...(record.style ?? {}),
   161→   161→        ...(width !== undefined ? { width } : {}),
   162→   162→        ...(height !== undefined ? { height } : {}),
   163→   163→      },
   164→   164→      ...(width !== undefined ? { width } : {}),
   165→   165→      ...(height !== undefined ? { height } : {}),
   166→   166→      // Deliberately no `measured` — it belongs to the exporting session and
   167→   167→      // React Flow re-measures the node on mount.
   168→   168→      selected: false,
   169→   169→      dragging: false,
   170→   170→      resizing: false,
   171→   171→    } as unknown as Node<ShapeData>
   172→   172→  })
   173→   173→  const nodeIds = new Set(nodes.map((n) => n.id))
   174→   174→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   175→   175→    .filter(
   176→   176→      (e) =>
   177→   177→        e &&
   178→   178→        typeof e.source === "string" &&
   179→   179→        typeof e.target === "string" &&
   180→   180→        nodeIds.has(e.source) &&
   181→   181→        nodeIds.has(e.target) &&
   182→   182→        e.source !== e.target,
   183→   183→    )
   184→   184→    .map((e, i) => ({
   185→   185→      ...e,
   186→   186→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   187→   187→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   188→   188→      selected: false,
   189→   189→    })) as Edge[]
   190→   190→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   191→   191→  return { nodes, edges, viewport }
   192→   192→}
