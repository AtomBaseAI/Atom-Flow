     1→/**
     2→ * Orthogonal edge-routing engine behind the board's SmartEdge.
     3→ *
     4→ * Given every node's bounding box and the board's edge list, `routeEdge`
     5→ * decides:
     6→ *
     7→ *  1. WHICH sides of the source/target boxes the edge should attach to.
     8→ *     16 side combinations are scored geometrically — a combination whose
     9→ *     naive orthogonal route would pass through another node (or fold back
    10→ *     through its own endpoints) is heavily penalised, so e.g. an edge whose
    11→ *     straight corridor is blocked by a middle node reroutes over/under it
    12→ *     instead (A top + X top when B sits between them).
    13→ *
    14→ *  2. HOW parallel edges share the same border. Edges attaching to the same
    15→ *     node side are spread evenly along that border (sorted by where their
    16→ *     far end sits), and the free "corridor" lane of head-on routes is
    17→ *     shifted by the average of the two endpoint offsets — so an A→D / D→A
    18→ *     pair renders as two parallel lines with a constant gap instead of
    19→ *     overlapping.
    20→ *
    21→ *  3. THE PATH between the anchors. When the naive route is clean it is used
    22→ *     directly (fast path, same shape as before). Otherwise a coarse-grid A*
    23→ *     finds an orthogonal detour around the obstacles (inflated by a margin),
    24→ *     which keeps edges from cutting through unrelated nodes.
    25→ *
    26→ * Everything here is pure geometry (no React, no store) so it can be unit
    27→ * tested directly.
    28→ */
    29→

   104→function inflateRect(r: Rect, p: number): Rect {
   105→  return { x: r.x - p, y: r.y - p, width: r.width + 2 * p, height: r.height + 2 * p }

   108→function shrinkRect(r: Rect, p: number): Rect {
   109→  return { x: r.x + p, y: r.y + p, width: Math.max(1, r.width - 2 * p), height: Math.max(1, r.height - 2 * p) }

   112→export function rectCenter(r: Rect): Pt {
   113→  return { x: r.x + r.width / 2, y: r.y + r.height / 2 }

   116→/** Point on the box border for `side`, slid `along` px parallel to that side. */
   117→export function borderPoint(rect: Rect, side: Side4, along = 0): Pt {
   118→  switch (side) {
   119→    case "top":
   120→      return { x: rect.x + rect.width / 2 + along, y: rect.y }
   121→    case "bottom":
   122→      return { x: rect.x + rect.width / 2 + along, y: rect.y + rect.height }
   123→    case "left":
   124→      return { x: rect.x, y: rect.y + rect.height / 2 + along }
   125→    case "right":
   126→      return { x: rect.x + rect.width, y: rect.y + rect.height / 2 + along }
   127→  }
   128→}

   130→/** Anchor = border point pushed `ANCHOR_GAP` px outside the box. */
   131→export function anchorPoint(rect: Rect, side: Side4, along = 0): Pt {
   132→  const b = borderPoint(rect, side, along)
   133→  const d = OUTWARD[side]
   134→  return { x: b.x + d.x * ANCHOR_GAP, y: b.y + d.y * ANCHOR_GAP }
   135→}

   137→/** Liang–Barsky segment vs rect intersection (strict overlap). */
   138→export function segHitsRect(a: Pt, b: Pt, r: Rect): boolean {
   139→  let t0 = 0
   140→  let t1 = 1
   141→  const dx = b.x - a.x
   142→  const dy = b.y - a.y
   143→  const p = [-dx, dx, -dy, dy]
   144→  const q = [a.x - r.x, r.x + r.width - a.x, a.y - r.y, r.y + r.height - a.y]
   145→  for (let i = 0; i < 4; i++) {
   146→    if (p[i] === 0) {
   147→      if (q[i] < 0) return false
   148→    } else {
   149→      const t = q[i] / p[i]
   150→      if (p[i] < 0) {
   151→        if (t > t1) return false
   152→        if (t > t0) t0 = t
   153→      } else {
   154→        if (t < t0) return false
   155→        if (t < t1) t1 = t
   156→      }
   157→    }
   158→  }
   159→  return t1 > t0
   160→}

   162→function polylineHits(pts: Pt[], rects: Rect[]): boolean {
   163→  for (let i = 0; i + 1 < pts.length; i++) {
   164→    for (const r of rects) {
   165→      if (segHitsRect(pts[i], pts[i + 1], r)) return true
   166→    }
   167→  }
   168→  return false
   169→}
   170→
   171→function polyLength(pts: Pt[]): number {
   172→  let len = 0
   173→  for (let i = 0; i + 1 < pts.length; i++) {
   174→    len += Math.hypot(pts[i + 1].x - pts[i].x, pts[i + 1].y - pts[i].y)
   175→  }
   176→  return len
   177→}

   179→function dedupe(pts: Pt[]): Pt[] {
   180→  const out: Pt[] = []
   181→  for (const p of pts) {
   182→    const last = out[out.length - 1]
   183→    if (!last || Math.abs(last.x - p.x) > 0.01 || Math.abs(last.y - p.y) > 0.01) out.push(p)
   184→  }
   185→  return out
   186→}
   187→
   188→/** Merge collinear segments so routes stay tidy. */
   189→export function simplifyPath(pts: Pt[]): Pt[] {
   190→  const src = dedupe(pts)
   191→  if (src.length < 3) return src
   192→  const out: Pt[] = [src[0]]
   193→  for (let i = 1; i < src.length - 1; i++) {
   194→    const a = out[out.length - 1]
   195→    const b = src[i]
   196→    const c = src[i + 1]
   197→    const cross = (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)
   198→    const dot = (b.x - a.x) * (c.x - b.x) + (b.y - a.y) * (c.y - b.y)
   199→    // Collinear (cross ≈ 0). Keep the point on direction reversals (dot < 0)
   200→    // so folded routes don't get flattened into a straight overlap.
   201→    if (Math.abs(cross) < 0.01 && dot >= 0) continue
   202→    out.push(b)
   203→  }
   204→  out.push(src[src.length - 1])
   205→  return out
   206→}

   212→/**
   213→ * Builds the classic orthogonal connector between two anchors:
   214→ * stub out of the source, bend toward the target, stub into the target.
   215→ * `midShift` slides the free corridor lane (head-on / same-direction shapes)
   216→ * sideways so parallel edges can occupy distinct lanes.
   217→ */
   218→export function simpleRoute(p0: Pt, d0: Pt, p3: Pt, d3: Pt, stub = STUB, midShift = 0): Pt[] {
   219→  const s0 = { x: p0.x + d0.x * stub, y: p0.y + d0.y * stub }
   220→  const s1 = { x: p3.x + d3.x * stub, y: p3.y + d3.y * stub }
   221→  const pts: Pt[] = [p0, s0]
   222→  const h0 = d0.x !== 0
   223→  const h3 = d3.x !== 0
   224→
   225→  if (h0 && h3) {
   226→    if (d0.x === -d3.x) {
   227→      // Head-on (e.g. right → left): straight lane when aligned, Z-bend else.
   228→      const forward = d0.x > 0 ? s0.x <= s1.x : s0.x >= s1.x
   229→      if (forward) {
   230→        if (Math.abs(s0.y - s1.y) < 0.5) {
   231→          pts.push(s1)
   232→        } else {
   233→          const mx = (s0.x + s1.x) / 2 + midShift
   234→          pts.push({ x: mx, y: s0.y }, { x: mx, y: s1.y }, s1)
   235→        }
   236→      } else {
   237→        const mx = d0.x > 0 ? Math.max(s0.x, s1.x) + stub : Math.min(s0.x, s1.x) - stub
   238→        pts.push({ x: mx, y: s0.y }, { x: mx, y: s1.y }, s1)
   239→      }
   240→    } else {
   241→      // Same direction: swing around the far side (S/U shape).
   242→      const mx = d0.x > 0 ? Math.max(s0.x, s1.x) + stub : Math.min(s0.x, s1.x) - stub
   243→      pts.push({ x: mx, y: s0.y }, { x: mx, y: s1.y }, s1)
   244→    }
   245→  } else if (!h0 && !h3) {
   246→    if (d0.y === -d3.y) {
   247→      const forward = d0.y > 0 ? s0.y <= s1.y : s0.y >= s1.y
   248→      if (forward) {
   249→        if (Math.abs(s0.x - s1.x) < 0.5) {
   250→          pts.push(s1)
   251→        } else {
   252→          const my = (s0.y + s1.y) / 2 + midShift
   253→          pts.push({ x: s0.x, y: my }, { x: s1.x, y: my }, s1)
   254→        }
   255→      } else {
   256→        const my = d0.y > 0 ? Math.max(s0.y, s1.y) + stub : Math.min(s0.y, s1.y) - stub
   257→        pts.push({ x: s0.x, y: my }, { x: s1.x, y: my }, s1)
   258→      }
   259→    } else {
   260→      const my = d0.y > 0 ? Math.max(s0.y, s1.y) + stub : Math.min(s0.y, s1.y) - stub
   261→      pts.push({ x: s0.x, y: my }, { x: s1.x, y: my }, s1)
   262→    }
   263→  } else {
   264→    // Perpendicular stubs: single elbow where they meet.
   265→    pts.push(h0 ? { x: s1.x, y: s0.y } : { x: s0.x, y: s1.y }, s1)
   266→  }
   267→
   268→  pts.push(p3)
   269→  return simplifyPath(pts)
   270→}

   276→interface RankedCombo {
   277→  source: Side4
   278→  target: Side4
   279→  score: number
   280→  /** Naive route is free of obstacle hits. */
   281→  clean: boolean
   282→}
   283→
   284→function rankCombos(srcRect: Rect, tgtRect: Rect, othersInf: Rect[], ownShrunk: Rect[]): RankedCombo[] {
   285→  const sc = rectCenter(srcRect)
   286→  const tc = rectCenter(tgtRect)
   287→  const dxc = tc.x - sc.x
   288→  const dyc = tc.y - sc.y
   289→  const distC = Math.hypot(dxc, dyc) || 1
   290→  const toT = { x: dxc / distC, y: dyc / distC }
   291→  const toS = { x: -toT.x, y: -toT.y }
   292→
   293→  const combos: RankedCombo[] = []
   294→  for (const ss of SIDE_ORDER) {
   295→    for (const ts of SIDE_ORDER) {
   296→      const a0 = anchorPoint(srcRect, ss)
   297→      const a3 = anchorPoint(tgtRect, ts)
   298→      const pts = simpleRoute(a0, OUTWARD[ss], a3, OUTWARD[ts])
   299→      let score = polyLength(pts)
   300→      const clean =
   301→        !polylineHits(pts, othersInf) && !polylineHits(pts.slice(1, -1), ownShrunk)
   302→      if (!clean) score += BLOCKED_PENALTY
   303→      const d0 = OUTWARD[ss]
   304→      const d3 = OUTWARD[ts]
   305→      score += (1 - (d0.x * toT.x + d0.y * toT.y)) * FACING_WEIGHT
   306→      score += (1 - (d3.x * toS.x + d3.y * toS.y)) * FACING_WEIGHT
   307→      if (ss === ts) score += SAME_SIDE_PENALTY
   308→      // Deterministic tie-break; "top" is checked first so an over-the-top
   309→      // detour wins an otherwise tied reroute (matches expectations when a
   310→      // middle node blocks the direct corridor).
   311→      score += (SIDE_ORDER.indexOf(ss) + SIDE_ORDER.indexOf(ts)) * 0.01
   312→      combos.push({ source: ss, target: ts, score, clean })
   313→    }
   314→  }
   315→  combos.sort((a, b) => a.score - b.score)
   316→  return combos
   317→}
   318→
   319→export function chooseSides(srcRect: Rect, tgtRect: Rect, othersInf: Rect[], ownShrunk: Rect[]): Sides {
   320→  const best = rankCombos(srcRect, tgtRect, othersInf, ownShrunk)[0]
   321→  return { source: best.source, target: best.target }
   322→}

   328→/** Along-border span available for anchor sliding. */
   329→function halfFor(rect: Rect, side: Side4): number {
   330→  const span = side === "top" || side === "bottom" ? rect.width : rect.height
   331→  return Math.max(0, span / 2 - 12)
   332→}
   333→
   334→/** Absolute coordinate of an anchor along its border axis. */
   335→function borderV(rect: Rect, side: Side4, along: number): number {
   336→  const b = borderPoint(rect, side, along)
   337→  return side === "top" || side === "bottom" ? b.x : b.y
   338→}
   339→
   340→/**
   341→ * Offset (px, along the border) for this edge's attachment on `nodeId`/`side`.
   342→ * All edges attaching to the same border are sorted by where their far end
   343→ * sits (edge id as tie-break) and spread apart (compressed if the border is
   344→ * too small for the full spacing). Bidirectional pairs therefore get
   345→ * symmetric ±lanes on both ends.
   346→ */
   347→function attachmentOffset(
   348→  selfId: string,
   349→  nodeId: string,
   350→  side: Side4,
   351→  edges: EdgeRef[],
   352→  boxById: Map<string, Box>,
   353→  sidesFor: (e: EdgeRef) => Sides,
   354→): number {
   355→  const box = boxById.get(nodeId)
   356→  if (!box) return 0
   357→  const axis: "x" | "y" = side === "top" || side === "bottom" ? "x" : "y"
   358→  const peers: { id: string; key: number }[] = []
   359→  for (const e of edges) {
   360→    const s = sidesFor(e)
   361→    let role: "source" | "target" | null = null
   362→    if (e.source === nodeId && s.source === side) role = "source"
   363→    else if (e.target === nodeId && s.target === side) role = "target"
   364→    if (!role) continue
   365→    const otherId = role === "source" ? e.target : e.source
   366→    const other = boxById.get(otherId)
   367→    const oc = other ? rectCenter(other.rect) : { x: 0, y: 0 }
   368→    peers.push({ id: e.id, key: axis === "x" ? oc.x : oc.y })
   369→  }
   370→  if (peers.length === 0) return 0
   371→  peers.sort((a, b) => a.key - b.key || (a.id < b.id ? -1 : a.id > b.id ? 1 : 0))
   372→  const idx = peers.findIndex((p) => p.id === selfId)
   373→  if (idx < 0) return 0
   374→  // Keep the anchor on the border segment; compress the ladder when the
   375→  // border cannot fit the full spacing.
   376→  const half = halfFor(box.rect, side)
   377→  const spacing = peers.length > 1 ? Math.min(LANE_SPACING, (2 * half) / (peers.length - 1)) : LANE_SPACING
   378→  const raw = (idx - (peers.length - 1) / 2) * spacing
   379→  return clamp(raw, -half, half)
   380→}

   386→function intervalsOverlap(a: [number, number], b: [number, number]): boolean {
   387→  return a[0] <= b[1] && b[0] <= a[1]
   388→}

   390→/** u-range spanned by an edge's corridor between its anchor node and far node. */
   391→function corridorBand(edge: EdgeRef, anchorNodeId: string, side: Side4, boxById: Map<string, Box>): [number, number] {
   392→  const horizontalCorridor = side === "left" || side === "right"
   393→  const farId = edge.source === anchorNodeId ? edge.target : edge.source
   394→  const a = boxById.get(anchorNodeId)
   395→  const b = boxById.get(farId)
   396→  if (!a || !b) return [Infinity, -Infinity]
   397→  const lo = horizontalCorridor ? (r: Rect) => r.x : (r: Rect) => r.y
   398→  const hi = horizontalCorridor ? (r: Rect) => r.x + r.width : (r: Rect) => r.y + r.height
   399→  return [Math.min(lo(a.rect), lo(b.rect)) - ANCHOR_GAP, Math.max(hi(a.rect), hi(b.rect)) + ANCHOR_GAP]
   400→}
   401→
   402→/**
   403→ * Final approach segments into left/right borders run horizontally at the
   404→ * anchor's y (into top/bottom borders: vertically at its x). When two edges
   405→ * anchor at nearly the same coordinate on such borders and their corridors
   406→ * overlap, their approaches would render as one merged line. This slides own
   407→ * anchor along its border to the nearest clear slot. Deterministic from both
   408→ * sides: the edge with more sliding headroom (ties: higher id) is the one
   409→ * that moves, and everyone evaluates conflicts against original offsets.
   410→ */
   411→function deconflictAnchor(
   412→  self: EdgeRef,
   413→  nodeId: string,
   414→  side: Side4,
   415→  off: number,
   416→  edges: EdgeRef[],
   417→  boxById: Map<string, Box>,
   418→  sidesFor: (e: EdgeRef) => Sides,
   419→  offsetsFor: (e: EdgeRef) => { source: number; target: number },
   420→): number {
   421→  const box = boxById.get(nodeId)
   422→  if (!box) return off
   423→  const axis: "x" | "y" = side === "top" || side === "bottom" ? "x" : "y"
   424→  const myV = borderV(box.rect, side, off)
   425→  const myBand = corridorBand(self, nodeId, side, boxById)
   426→  const myHeadroom = halfFor(box.rect, side) - Math.abs(off)
   427→
   428→  const conflicts: { id: string; v: number; headroom: number }[] = []
   429→  for (const peer of edges) {
   430→    if (peer.id === self.id) continue
   431→    const ps = sidesFor(peer)
   432→    const pOffs = offsetsFor(peer)
   433→    const attachments: [string, Side4, number][] = [
   434→      [peer.source, ps.source, pOffs.source],
   435→      [peer.target, ps.target, pOffs.target],
   436→    ]
   437→    for (const [pNode, pSide, pOff] of attachments) {
   438→      const pAxis: "x" | "y" = pSide === "top" || pSide === "bottom" ? "x" : "y"
   439→      if (pAxis !== axis) continue
   440→      const pBox = boxById.get(pNode)
   441→      if (!pBox) continue
   442→      const pv = borderV(pBox.rect, pSide, pOff)
   443→      if (Math.abs(pv - myV) >= ANCHOR_CLEAR) continue
   444→      if (!intervalsOverlap(myBand, corridorBand(peer, pNode, pSide, boxById))) continue
   445→      conflicts.push({ id: peer.id, v: pv, headroom: halfFor(pBox.rect, pSide) - Math.abs(pOff) })
   446→    }
   447→  }
   448→  if (conflicts.length === 0) return off
   449→
   450→  const mustMove = conflicts.some(
   451→    (c) => myHeadroom > c.headroom || (myHeadroom === c.headroom && self.id > c.id),
   452→  )
   453→  if (!mustMove) return off
   454→
   455→  const avoid = conflicts.map((c) => c.v)
   456→  const half = halfFor(box.rect, side)
   457→  for (let step = 1; step <= 6; step++) {
   458→    for (const dir of [1, -1]) {
   459→      const candidate = off + dir * step * ANCHOR_CLEAR
   460→      if (Math.abs(candidate) > half) continue
   461→      const cv = borderV(box.rect, side, candidate)
   462→      if (avoid.every((v) => Math.abs(cv - v) >= ANCHOR_CLEAR)) return candidate
   463→    }
   464→  }
   465→  return off
   466→}

   468→/**
   469→ * Bidirectional pairs run two head-on corridors over the same span. Each
   470→ * edge slides its corridor lane half a lane-width toward its own target so
   471→ * the two routes become parallel lines instead of interweaving.
   472→ */
   473→function pairLaneShift(
   474→  self: EdgeRef,
   475→  sides: Sides,
   476→  edges: EdgeRef[],
   477→  boxById: Map<string, Box>,
   478→  sidesFor: (e: EdgeRef) => Sides,
   479→): number {
   480→  const reverse = edges.find((e) => e.id !== self.id && e.source === self.target && e.target === self.source)
   481→  if (!reverse) return 0
   482→  // Only meaningful for head-on combos (opposite sides).
   483→  const d0 = OUTWARD[sides.source]
   484→  const d3 = OUTWARD[sides.target]
   485→  if (d0.x !== -d3.x || d0.y !== -d3.y) return 0
   486→  const rs = sidesFor(reverse)
   487→  if (rs.source !== sides.target || rs.target !== sides.source) return 0
   488→  const sc = rectCenter(boxById.get(self.source)!.rect)
   489→  const tc = rectCenter(boxById.get(self.target)!.rect)
   490→  const horizontal = sides.source === "left" || sides.source === "right"
   491→  const dir = horizontal ? Math.sign(tc.x - sc.x) : Math.sign(tc.y - sc.y)
   492→  return (LANE_SPACING / 2) * dir
   493→}

   495→// ---------------------------------------------------------------------------
   496→// Grid A* fallback for fully blocked corridors
   497→// ---------------------------------------------------------------------------
   498→
   499→/**
   500→ * Orthogonal A* on a coarse grid. Obstacles must already be inflated. Returns
   501→ * interior waypoints from `s0` to `s1` (excluding the endpoints), or null.
   502→ */
   503→function astarRoute(s0: Pt, s1: Pt, obstacles: Rect[]): Pt[] | null {
   504→  const cellBase = 16
   505→  let minX = Math.min(s0.x, s1.x)
   506→  let minY = Math.min(s0.y, s1.y)
   507→  let maxX = Math.max(s0.x, s1.x)
   508→  let maxY = Math.max(s0.y, s1.y)
   509→  for (const o of obstacles) {
   510→    minX = Math.min(minX, o.x)
   511→    minY = Math.min(minY, o.y)
   512→    maxX = Math.max(maxX, o.x + o.width)
   513→    maxY = Math.max(maxY, o.y + o.height)
   514→  }
   515→  const margin = cellBase * 3
   516→  minX -= margin
   517→  minY -= margin
   518→  maxX += margin
   519→  maxY += margin
   520→
   521→  let cell = cellBase
   522→  let cols = Math.ceil((maxX - minX) / cell) + 1
   523→  let rows = Math.ceil((maxY - minY) / cell) + 1
   524→  while (cols * rows > 60000) {
   525→    cell *= 2
   526→    cols = Math.ceil((maxX - minX) / cell) + 1
   527→    rows = Math.ceil((maxY - minY) / cell) + 1
   528→  }
   529→
   530→  const size = cols * rows
   531→  const blocked = new Uint8Array(size)
   532→  const cellIdx = (c: number, r: number) => r * cols + c
   533→  for (let r = 0; r < rows; r++) {
   534→    const y = minY + r * cell
   535→    for (let c = 0; c < cols; c++) {
   536→      const x = minX + c * cell
   537→      for (const o of obstacles) {
   538→        if (x < o.x + o.width + 2 && x + cell > o.x - 2 && y < o.y + o.height + 2 && y + cell > o.y - 2) {
   539→          blocked[cellIdx(c, r)] = 1
   540→          break
   541→        }
   542→      }
   543→    }
   544→  }
   545→
   546→  const toCell = (v: number, lo: number) => clamp(Math.round((v - lo) / cell), 0, cols - 1)
   547→  const clearNear = (p: Pt) => {
   548→    const c0 = clamp(Math.round((p.x - minX) / cell), 0, cols - 1)
   549→    const r0 = clamp(Math.round((p.y - minY) / cell), 0, rows - 1)
   550→    for (let dr = -1; dr <= 1; dr++) {
   551→      for (let dc = -1; dc <= 1; dc++) {
   552→        const c = c0 + dc
   553→        const r = r0 + dr
   554→        if (c >= 0 && c < cols && r >= 0 && r < rows) blocked[cellIdx(c, r)] = 0
   555→      }
   556→    }
   557→  }
   558→  clearNear(s0)
   559→  clearNear(s1)
   560→
   561→  const startC = toCell(s0.x, minX)
   562→  const startR = toCell(s0.y, minY)
   563→  const goalC = toCell(s1.x, minX)
   564→  const goalR = toCell(s1.y, minY)
   565→  const startIdx = cellIdx(startC, startR)
   566→  const goalIdx = cellIdx(goalC, goalR)
   567→
   568→  // State = cell × direction (0:+x, 1:-x, 2:+y, 3:-y).
   569→  const DIRS: Pt[] = [
   570→    { x: 1, y: 0 },
   571→    { x: -1, y: 0 },
   572→    { x: 0, y: 1 },
   573→    { x: 0, y: -1 },
   574→  ]
   575→  const TURN_COST = 0.4
   576→  const states = size * 4
   577→  const gScore = new Float64Array(states).fill(Infinity)
   578→  const parent = new Int32Array(states).fill(-1)
   579→  const closed = new Uint8Array(states)
   580→
   581→  const h = (c: number, r: number) => Math.abs(c - goalC) + Math.abs(r - goalR)
   582→
   583→  // Minimal binary heap over (f, state).
   584→  const heapF: number[] = []
   585→  const heapS: number[] = []
   586→  const push = (f: number, s: number) => {
   587→    heapF.push(f)
   588→    heapS.push(s)
   589→    let i = heapF.length - 1
   590→    while (i > 0) {
   591→      const p = (i - 1) >> 1
   592→      if (heapF[p] <= heapF[i]) break
   593→      ;[heapF[p], heapF[i]] = [heapF[i], heapF[p]]
   594→      ;[heapS[p], heapS[i]] = [heapS[i], heapS[p]]
   595→      i = p
   596→    }
   597→  }
   598→  const pop = (): number => {
   599→    const top = heapS[0]
   600→    const lf = heapF.pop() as number
   601→    const ls = heapS.pop() as number
   602→    if (heapF.length > 0) {
   603→      heapF[0] = lf
   604→      heapS[0] = ls
   605→      let i = 0
   606→      for (;;) {
   607→        const l = 2 * i + 1
   608→        const r = l + 1
   609→        let m = i
   610→        if (l < heapF.length && heapF[l] < heapF[m]) m = l
   611→        if (r < heapF.length && heapF[r] < heapF[m]) m = r
   612→        if (m === i) break
   613→        ;[heapF[m], heapF[i]] = [heapF[i], heapF[m]]
   614→        ;[heapS[m], heapS[i]] = [heapS[i], heapS[m]]
   615→        i = m
   616→      }
   617→    }
   618→    return top
   619→  }
   620→
   621→  for (let d = 0; d < 4; d++) {
   622→    gScore[startIdx * 4 + d] = 0
   623→    push(h(startC, startR), startIdx * 4 + d)
   624→  }
   625→
   626→  let goalState = -1
   627→  while (heapF.length > 0) {
   628→    const state = pop()
   629→    if (closed[state]) continue
   630→    closed[state] = 1
   631→    const cellId = Math.floor(state / 4)
   632→    const dir = state % 4
   633→    if (cellId === goalIdx) {
   634→      goalState = state
   635→      break
   636→    }
   637→    const c = cellId % cols
   638→    const r = Math.floor(cellId / cols)
   639→    const g = gScore[state]
   640→    for (let nd = 0; nd < 4; nd++) {
   641→      const dc = DIRS[nd].x
   642→      const dr = DIRS[nd].y
   643→      const nc = c + dc
   644→      const nr = r + dr
   645→      if (nc < 0 || nc >= cols || nr < 0 || nr >= rows) continue
   646→      const nCell = cellIdx(nc, nr)
   647→      if (blocked[nCell]) continue
   648→      const ng = g + 1 + (nd === dir ? 0 : TURN_COST)
   649→      const nState = nCell * 4 + nd
   650→      if (ng < gScore[nState] - 1e-9) {
   651→        gScore[nState] = ng
   652→        parent[nState] = state
   653→        push(ng + h(nc, nr), nState)
   654→      }
   655→    }
   656→  }
   657→
   658→  if (goalState < 0) return null
   659→
   660→  const rev: Pt[] = []
   661→  let cur = goalState
   662→  while (cur >= 0) {
   663→    const cellId = Math.floor(cur / 4)
   664→    const c = cellId % cols
   665→    const r = Math.floor(cellId / cols)
   666→    rev.push({ x: minX + c * cell, y: minY + r * cell })
   667→    cur = parent[cur]
   668→  }
   669→  rev.reverse()
   670→  // rev[0] is the start cell, rev[last] the goal cell. Align the first/last
   671→  // interior points with the stub axes so no jog appears next to the anchor.
   672→  const pts = rev.slice(1, -1)
   673→  if (pts.length > 0) {
   674→    const first = pts[0]
   675→    const startDirIsX = Math.abs(first.x - rev[0].x) >= Math.abs(first.y - rev[0].y)
   676→    if (startDirIsX) first.y = s0.y
   677→    else first.x = s0.x
   678→    const last = pts[pts.length - 1]
   679→    const endDirIsX = Math.abs(rev[rev.length - 1].x - last.x) >= Math.abs(rev[rev.length - 1].y - last.y)
   680→    if (endDirIsX) last.y = s1.y
   681→    else last.x = s1.x
   682→  }
   683→  return simplifyPath(pts)
   684→}

   691→  edgeId: string
   692→  sourceId: string
   693→  targetId: string
   694→  /** Every node's bounding box on the board. */
   695→  boxes: Box[]
   696→  /** Every edge on the board (used for lane bundling). */

   700→/**
   701→ * Computes the smart route for one edge:
   702→ * sides → lane offsets → path (fast naive route, A* detour as fallback).
   703→ */
   704→export function routeEdge(input: RouteInput): RouteResult | null {
   705→  const { edgeId, sourceId, targetId, boxes, edges } = input
   706→  const boxById = new Map<string, Box>()
   707→  for (const b of boxes) boxById.set(b.id, b)
   708→  const srcBox = boxById.get(sourceId)
   709→  const tgtBox = boxById.get(targetId)
   710→  if (!srcBox || !tgtBox) return null
   711→
   712→  const sidesCache = new Map<string, Sides>()
   713→  const sidesFor = (e: EdgeRef): Sides => {
   714→    const hit = sidesCache.get(e.id)
   715→    if (hit) return hit
   716→    const a = boxById.get(e.source)
   717→    const b = boxById.get(e.target)
   718→    let result: Sides
   719→    if (!a || !b || (a.rect.width <= 0 || a.rect.height <= 0 || b.rect.width <= 0 || b.rect.height <= 0)) {
   720→      result = { source: "right", target: "left" }
   721→    } else {
   722→      const othersInf = boxes
   723→        .filter((bx) => bx.id !== e.source && bx.id !== e.target)
   724→        .map((bx) => inflateRect(bx.rect, OBSTACLE_PAD))
   725→      const ownShrunk = [shrinkRect(a.rect, 1), shrinkRect(b.rect, 1)]
   726→      result = chooseSides(a.rect, b.rect, othersInf, ownShrunk)
   727→    }
   728→    sidesCache.set(e.id, result)
   729→    return result
   730→  }
   731→
   732→  const offsCache = new Map<string, { source: number; target: number }>()
   733→  const offsetsFor = (e: EdgeRef): { source: number; target: number } => {
   734→    const hit = offsCache.get(e.id)
   735→    if (hit) return hit
   736→    const s = sidesFor(e)
   737→    const result = {
   738→      source: attachmentOffset(e.id, e.source, s.source, edges, boxById, sidesFor),
   739→      target: attachmentOffset(e.id, e.target, s.target, edges, boxById, sidesFor),
   740→    }
   741→    offsCache.set(e.id, result)
   742→    return result
   743→  }
   744→
   745→  const self: EdgeRef = { id: edgeId, source: sourceId, target: targetId }
   746→  const sides = sidesFor(self)
   747→
   748→  // Spread on shared borders.
   749→  const baseOffs = offsetsFor(self)
   750→  let srcOff = baseOffs.source
   751→  let tgtOff = baseOffs.target
   752→
   753→  // Mirrored *perpendicular* pairs (one edge L-shaped one way, its reverse
   754→  // the other way) interweave unless their anchor order at one shared border
   755→  // is reversed. The border to fix is the leader edge's (smaller id) target
   756→  // border — both edges compute the same leader, so they agree.
   757→  const reverse = edges.find((e) => e.id !== edgeId && e.source === targetId && e.target === sourceId)
   758→  if (reverse) {
   759→    const rs = sidesFor(reverse)
   760→    const mirrored = rs.source === sides.target && rs.target === sides.source
   761→    const d0 = OUTWARD[sides.source]
   762→    const d3 = OUTWARD[sides.target]
   763→    const headOn = d0.x === -d3.x && d0.y === -d3.y
   764→    const perpendicular = d0.x !== 0 !== (d3.x !== 0)
   765→    if (mirrored && perpendicular) {
   766→      const leader = self.id < reverse.id ? self : reverse
   767→      const rOffs = offsetsFor(reverse)
   768→      if (leader.id === self.id) {
   769→        // Reverse the order at my target border (partner's source border).
   770→        const sigma = -Math.sign(srcOff - rOffs.target || (self.id < reverse.id ? 1 : -1))
   771→        const desired = sigma > 0 ? Math.max(tgtOff, rOffs.source) : Math.min(tgtOff, rOffs.source)
   772→        if (desired !== tgtOff) tgtOff = desired
   773→      } else {
   774→        // Leader's target border is my source border: I take the slot that
   775→        // keeps the two borders' orders opposed (leader picks the other one).
   776→        const sigma = -Math.sign(rOffs.source - tgtOff || (reverse.id < self.id ? 1 : -1))
   777→        const desired = sigma > 0 ? Math.min(srcOff, rOffs.target) : Math.max(srcOff, rOffs.target)
   778→        if (desired !== srcOff) srcOff = desired
   779→      }
   780→    }
   781→  }
   782→
   783→  // Slide anchors whose parallel approaches would collide with another
   784→  // edge's approach.
   785→  srcOff = deconflictAnchor(self, sourceId, sides.source, srcOff, edges, boxById, sidesFor, offsetsFor)
   786→  tgtOff = deconflictAnchor(self, targetId, sides.target, tgtOff, edges, boxById, sidesFor, offsetsFor)
   787→
   788→  // Corridor lane: counter-shift by the endpoint spread plus the bidirectional
   789→  // pair bias so parallel edges keep a constant gap along the whole run.
   790→  const midShift = (srcOff - tgtOff) / 2 + pairLaneShift(self, sides, edges, boxById, sidesFor)
   791→
   792→  const othersInf = boxes
   793→    .filter((bx) => bx.id !== sourceId && bx.id !== targetId)
   794→    .map((bx) => inflateRect(bx.rect, OBSTACLE_PAD))
   795→  const ownShrunk = [shrinkRect(srcBox.rect, 1), shrinkRect(tgtBox.rect, 1)]
   796→
   797→  const isClean = (pts: Pt[]) => !polylineHits(pts, othersInf) && !polylineHits(pts.slice(1, -1), ownShrunk)
   798→
   799→  const buildSimple = (ss: Side4, ts: Side4, shift: number): Pt[] => {
   800→    const a0 = anchorPoint(srcBox.rect, ss, ss === sides.source ? srcOff : 0)
   801→    const a3 = anchorPoint(tgtBox.rect, ts, ts === sides.target ? tgtOff : 0)
   802→    return simpleRoute(a0, OUTWARD[ss], a3, OUTWARD[ts], STUB, shift)
   803→  }
   804→
   805→  // 1) Fast path: first clean combo (best-scored) already avoids everything.
   806→  //    If the full lane shift pushes the corridor into an obstacle, degrade
   807→  //    it gracefully (half, then zero) — never flip its sign, which would
   808→  //    re-introduce pair interweaving.
   809→  const ranked = rankCombos(srcBox.rect, tgtBox.rect, othersInf, ownShrunk)
   810→  for (const combo of ranked) {
   811→    if (!combo.clean) continue
   812→    for (const shift of [midShift, midShift / 2, 0]) {
   813→      const pts = buildSimple(combo.source, combo.target, shift)
   814→      if (isClean(pts)) {
   815→        return { points: pts, sourceSide: combo.source, targetSide: combo.target }
   816→      }
   817→    }
   818→  }
   819→
   820→  // 2) Nothing clean: A* detours for the most promising combos.
   821→  let best: { pts: Pt[]; cost: number; ss: Side4; ts: Side4 } | null = null
   822→  for (const combo of ranked.slice(0, ASTAR_CANDIDATES)) {
   823→    const a0 = anchorPoint(srcBox.rect, combo.source, combo.source === sides.source ? srcOff : 0)
   824→    const a3 = anchorPoint(tgtBox.rect, combo.target, combo.target === sides.target ? tgtOff : 0)
   825→    const s0 = { x: a0.x + OUTWARD[combo.source].x * STUB, y: a0.y + OUTWARD[combo.source].y * STUB }
   826→    const s1 = { x: a3.x + OUTWARD[combo.target].x * STUB, y: a3.y + OUTWARD[combo.target].y * STUB }
   827→    const astarObstacles = [...othersInf, ...ownShrunk]
   828→    const mid = astarRoute(s0, s1, astarObstacles)
   829→    if (!mid) continue
   830→    const pts = simplifyPath([a0, s0, ...mid, s1, a3])
   831→    const cost = polyLength(pts) + countTurns(pts) * 8
   832→    if (!best || cost < best.cost) best = { pts, cost, ss: combo.source, ts: combo.target }
   833→  }
   834→  if (best) return { points: best.pts, sourceSide: best.ss, targetSide: best.ts }
   835→
   836→  // 3) Last resort: best combo's naive route even if it grazes something.
   837→  const fallback = ranked[0]
   838→  return {
   839→    points: buildSimple(fallback.source, fallback.target, midShift),
   840→    sourceSide: fallback.source,
   841→    targetSide: fallback.target,
   842→  }
   843→}
   844→
   845→function countTurns(pts: Pt[]): number {
   846→  let turns = 0
   847→  for (let i = 1; i + 1 < pts.length; i++) {
   848→    const a = pts[i - 1]
   849→    const b = pts[i]
   850→    const c = pts[i + 1]
   851→    if (Math.abs((b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x)) > 0.01) turns++
   852→  }
   853→  return turns
   854→}

   856→// ---------------------------------------------------------------------------
   857→// SVG path with rounded corners
   858→// ---------------------------------------------------------------------------
   859→
   860→/** Renders a polyline as an SVG path with quadratic-rounded corners. */
   861→export function roundedPath(points: Pt[], radius = CORNER_RADIUS): string {
   862→  const f = (v: number) => {
   863→    const r = Math.round(v * 100) / 100
   864→    return Number.isFinite(r) ? String(r) : "0"
   865→  }
   866→  if (points.length === 0) return ""
   867→  if (points.length === 1) return `M ${f(points[0].x)} ${f(points[0].y)}`
   868→  let d = `M ${f(points[0].x)} ${f(points[0].y)}`
   869→  for (let i = 1; i < points.length - 1; i++) {
   870→    const prev = points[i - 1]
   871→    const cur = points[i]
   872→    const next = points[i + 1]
   873→    const l1 = Math.hypot(cur.x - prev.x, cur.y - prev.y)
   874→    const l2 = Math.hypot(next.x - cur.x, next.y - cur.y)
   875→    if (l1 < 0.01 || l2 < 0.01) continue
   876→    const r = Math.min(radius, l1 / 2, l2 / 2)
   877→    const a = { x: cur.x - ((cur.x - prev.x) / l1) * r, y: cur.y - ((cur.y - prev.y) / l1) * r }
   878→    const b = { x: cur.x + ((next.x - cur.x) / l2) * r, y: cur.y + ((next.y - cur.y) / l2) * r }
   879→    d += ` L ${f(a.x)} ${f(a.y)} Q ${f(cur.x)} ${f(cur.y)} ${f(b.x)} ${f(b.y)}`
   880→  }
   881→  const last = points[points.length - 1]
   882→  d += ` L ${f(last.x)} ${f(last.y)}`
   883→  return d
   884→}
