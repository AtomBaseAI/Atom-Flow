   132→   132→function IconPreview({
   133→   133→  src,
   134→   134→  lucideComp,
   135→   135→  label,
   136→   136→  style,
   137→   137→}: {
   138→   138→  src: string
   139→   139→  lucideComp?: React.ComponentType<{ className?: string; style?: React.CSSProperties }>
   140→   140→  label: string
   141→   141→  style: IconStyleId
   142→   142→}) {
   143→   143→  const imgSrc = src.startsWith("lucide:") ? "" : src
   144→   144→  const Glyph = lucideComp
   145→   145→  const borderColor = "var(--af-node-border)"
   146→   146→  const textColor = "var(--af-node-text)"
   147→   147→
   148→   148→  if (style === "frame") {
   149→   149→    return (
   150→   150→      <div
   151→   151→        className="overflow-hidden rounded-[4px] shadow-sm"
   152→   152→        style={{ width: 80, height: 60, background: "var(--af-frame-bg)", border: `1px solid ${borderColor}` }}
   153→   153→      >
   154→   154→        <div className="flex items-center justify-between gap-1 border-b border-black/10 px-1 py-0.5 dark:border-white/15">
   155→   155→          <span
   156→   156→            className="min-w-0 flex-1 truncate text-left text-[7px] font-semibold leading-[1.2]"
   157→   157→            style={{ color: textColor }}
   158→   158→          >
   159→   159→            {label}
   160→   160→          </span>
   161→   161→          {Glyph ? (
   162→   162→            <Glyph className="h-2.5 w-2.5" style={{ color: borderColor }} aria-hidden />
   163→   163→          ) : imgSrc ? (
   164→   164→            <img src={imgSrc} alt="" className="h-3 w-3 object-contain" draggable={false} />
   165→   165→          ) : null}
   166→   166→        </div>
   167→   167→        <div />
   168→   168→      </div>
   169→   169→    )
   170→   170→  }
   171→   171→
   172→   172→  if (style === "icon") {
   173→   173→    return (
   174→   174→      <div
   175→   175→        className="flex items-center justify-center rounded-[2px]"
   176→   176→        style={{ width: 96, height: 96, border: `1px solid ${borderColor}` }}
   177→   177→      >
   178→   178→        {Glyph ? (
   179→   179→          <Glyph className="h-[80%] w-[80%]" style={{ color: borderColor }} aria-hidden />
   180→   180→        ) : (
   181→   181→          <img src={imgSrc || "/placeholder.svg"} alt="" className="h-[85%] w-[85%] object-contain" draggable={false} />
   182→   182→        )}
   183→   183→      </div>
   184→   184→    )
   185→   185→  }
   186→   186→
   187→   187→  // labeled — icon with bottom text strip (classic)
   188→   188→  return (
   189→   189→    <div
   190→   190→      className="flex flex-col overflow-hidden rounded-[2px]"
   191→   191→      style={{ width: 96, height: 96, border: `1px solid ${borderColor}` }}
   192→   192→    >
   193→   193→      <div className="flex min-h-0 flex-1 items-center justify-center p-1">
   194→   194→        {Glyph ? (
   195→   195→          <Glyph className="h-[85%] w-[85%]" style={{ color: borderColor }} aria-hidden />
   196→   196→        ) : (
   197→   197→          <img src={imgSrc || "/placeholder.svg"} alt="" className="h-[85%] w-[85%] object-contain" draggable={false} />
   198→   198→        )}
   199→   199→      </div>
   200→   200→      <div className="flex min-h-[24px] flex-none items-center justify-center border-t border-black/10 px-1 dark:border-white/15">
   201→   201→        <span className="w-full truncate text-center text-[8px] font-bold leading-tight" style={{ color: textColor }}>
   202→   202→          {label}
   203→   203→        </span>
   204→   204→      </div>
   205→   205→    </div>
   206→   206→  )
   207→   207→}
