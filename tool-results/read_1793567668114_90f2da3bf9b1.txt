     1→/**
     2→ * Orthogonal edge-routing engine behind the board's SmartEdge.
     3→ *
     4→ * Given every node's bounding box and the board's edge list, `routeEdge`
     5→ * decides:
     6→ *
     7→ *  1. WHICH sides of the source/target boxes the edge should attach to.
     8→ *     16 side combinations are scored geometrically — a combination whose
     9→ *     naive orthogonal route would pass through another node (or fold back
    10→ *     through its own endpoints) is heavily penalised, so e.g. an edge whose
    11→ *     straight corridor is blocked by a middle node reroutes over/under it
    12→ *     instead (A top + X top when B sits between them).
    13→ *
    14→ *  2. HOW parallel edges share the same border. Edges attaching to the same
    15→ *     node side are spread evenly along that border (sorted by where their
    16→ *     far end sits), and the free "corridor" lane of head-on routes is
    17→ *     shifted by the average of the two endpoint offsets — so an A→D / D→A
    18→ *     pair renders as two parallel lines with a constant gap instead of
    19→ *     overlapping.
    20→ *
    21→ *  3. THE PATH between the anchors. When the naive route is clean it is used
    22→ *     directly (fast path, same shape as before). Otherwise a coarse-grid A*
    23→ *     finds an orthogonal detour around the obstacles (inflated by a margin),
    24→ *     which keeps edges from cutting through unrelated nodes.
    25→ *
    26→ * Stability guarantees (the "random paths" fixes):
    27→ *
    28→ *  - CONTAINMENT-AWARE OBSTACLES. A node that fully contains an endpoint
    29→ *    (a group/container rectangle drawn around icon nodes, for example) is
    30→ *    never an obstacle for that endpoint's edges — a child edge has to cross
    31→ *    the container border, and treating the container as an obstacle used to
    32→ *    force every side combo to score as blocked, producing squiggly A*
    33→ *    escape paths that changed with every unrelated drag.
    34→ *
    35→ *  - SIDE-CHOICE HYSTERESIS. The chosen attachment sides are remembered per
    36→ *    edge; while the remembered combo keeps a clean route it is retained
    37→ *    unless the best combo becomes substantially shorter. Near-tied combos
    38→ *    therefore stop flapping when an unrelated node moves.
    39→ *
    40→ *  - DIRECTION-CONSTRAINED A*. Detours leave the source along its stub and
    41→ *    arrive at the target along its stub, and folded (back-and-forth)
    42→ *    segments are pruned, so detours never render as hooks or loops at the
    43→ *    anchors.
    44→ *
    45→ * Everything here is pure geometry (no React, no store) so it can be unit
    46→ * tested directly.
    47→ */
    48→

   131→/** True when `outer` fully encloses `inner` (with a small tolerance). */
   132→export function containsRect(outer: Rect, inner: Rect, tol = 8): boolean {
   133→  if (outer.width <= 0 || outer.height <= 0 || inner.width <= 0 || inner.height <= 0) return false
   134→  // A container must be strictly larger than what it contains — two separate
   135→  // nodes of the same size never contain each other.
   136→  if (outer.width * outer.height <= inner.width * inner.height) return false
   137→  return (
   138→    inner.x >= outer.x - tol &&
   139→    inner.y >= outer.y - tol &&
   140→    inner.x + inner.width <= outer.x + outer.width + tol &&
   141→    inner.y + inner.height <= outer.y + outer.height + tol
   142→  )
   143→}

   145→/**
   146→ * Node ids that must NOT act as obstacles for an edge between `sourceId` and
   147→ * `targetId`. Beyond the endpoints themselves this excludes visual
   148→ * container ancestors: a node whose rect fully contains an endpoint (e.g. a
   149→ * "group" rectangle drawn around several icon nodes). Edges attached to a
   150→ * container's children have to cross the container border, so treating the
   151→ * container as an obstacle would block every side combination and produce
   152→ * squiggly A* escape paths. Children of an endpoint stay obstacles on
   153→ * purpose — an edge leaving a container should still detour around the
   154→ * icons it visually passes over.
   155→ */
   156→function obstacleExclusionsFor(
   157→  sourceId: string,
   158→  targetId: string,
   159→  boxes: Box[],
   160→  boxById: Map<string, Box>,
   161→): Set<string> {
   162→  const excluded = new Set<string>([sourceId, targetId])
   163→  const src = boxById.get(sourceId)
   164→  const tgt = boxById.get(targetId)
   165→  for (const bx of boxes) {
   166→    if (excluded.has(bx.id)) continue
   167→    if (src && containsRect(bx.rect, src.rect)) {
   168→      excluded.add(bx.id)
   169→      continue
   170→    }
   171→    if (tgt && containsRect(bx.rect, tgt.rect)) {
   172→      excluded.add(bx.id)
   173→    }
   174→  }
   175→  return excluded
   176→}

   182→/** Point on the box border for `side`, slid `along` px parallel to that side. */
   183→export function borderPoint(rect: Rect, side: Side4, along = 0): Pt {
   184→  switch (side) {
   185→    case "top":
   186→      return { x: rect.x + rect.width / 2 + along, y: rect.y }
   187→    case "bottom":
   188→      return { x: rect.x + rect.width / 2 + along, y: rect.y + rect.height }
   189→    case "left":
   190→      return { x: rect.x, y: rect.y + rect.height / 2 + along }
   191→    case "right":
   192→      return { x: rect.x + rect.width, y: rect.y + rect.height / 2 + along }
   193→  }
   194→}

   203→/** Liang–Barsky segment vs rect intersection (strict overlap). */
   204→export function segHitsRect(a: Pt, b: Pt, r: Rect): boolean {
   205→  let t0 = 0
   206→  let t1 = 1
   207→  const dx = b.x - a.x
   208→  const dy = b.y - a.y
   209→  const p = [-dx, dx, -dy, dy]
   210→  const q = [a.x - r.x, r.x + r.width - a.x, a.y - r.y, r.y + r.height - a.y]
   211→  for (let i = 0; i < 4; i++) {
   212→    if (p[i] === 0) {
   213→      if (q[i] < 0) return false
   214→    } else {
   215→      const t = q[i] / p[i]
   216→      if (p[i] < 0) {
   217→        if (t > t1) return false
   218→        if (t > t0) t0 = t
   219→      } else {
   220→        if (t < t0) return false
   221→        if (t < t1) t1 = t
   222→      }
   223→    }
   224→  }
   225→  return t1 > t0
   226→}

   228→function polylineHits(pts: Pt[], rects: Rect[]): boolean {
   229→  for (let i = 0; i + 1 < pts.length; i++) {
   230→    for (const r of rects) {
   231→      if (segHitsRect(pts[i], pts[i + 1], r)) return true
   232→    }
   233→  }
   234→  return false
   235→}

   237→function polyLength(pts: Pt[]): number {
   238→  let len = 0
   239→  for (let i = 0; i + 1 < pts.length; i++) {
   240→    len += Math.hypot(pts[i + 1].x - pts[i].x, pts[i + 1].y - pts[i].y)
   241→  }
   242→  return len
   243→}

   245→function dedupe(pts: Pt[]): Pt[] {
   246→  const out: Pt[] = []
   247→  for (const p of pts) {
   248→    const last = out[out.length - 1]
   249→    if (!last || Math.abs(last.x - p.x) > 0.01 || Math.abs(last.y - p.y) > 0.01) out.push(p)
   250→  }
   251→  return out
   252→}

   254→/** Merge collinear segments so routes stay tidy. */
   255→export function simplifyPath(pts: Pt[]): Pt[] {
   256→  const src = dedupe(pts)
   257→  if (src.length < 3) return src
   258→  const out: Pt[] = [src[0]]
   259→  for (let i = 1; i < src.length - 1; i++) {
   260→    const a = out[out.length - 1]
   261→    const b = src[i]
   262→    const c = src[i + 1]
   263→    const cross = (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)
   264→    const dot = (b.x - a.x) * (c.x - b.x) + (b.y - a.y) * (c.y - b.y)
   265→    // Collinear (cross ≈ 0). Keep the point on direction reversals (dot < 0)
   266→    // so folded routes don't get flattened into a straight overlap.
   267→    if (Math.abs(cross) < 0.01 && dot >= 0) continue
   268→    out.push(b)
   269→  }
   270→  out.push(src[src.length - 1])
   271→  return out
   272→}
   273→
   274→/**
   275→ * Cancels folded (overlapping back-and-forth) segments, e.g. a route that
   276→ * runs right past an anchor and then doubles back over itself. Removing the
   277→ * corner point of each fold shortens the drawn overlap while keeping the
   278→ * exact endpoints, so detours never render as scribbles. Non-orthogonal
   279→ * input is returned untouched.
   280→ */
   281→export function pruneBacktracks(pts: Pt[]): Pt[] {
   282→  for (let i = 0; i + 1 < pts.length; i++) {
   283→    const dx = Math.abs(pts[i + 1].x - pts[i].x)
   284→    const dy = Math.abs(pts[i + 1].y - pts[i].y)
   285→    if (dx > 0.01 && dy > 0.01) return pts
   286→  }
   287→  const out = pts.map((p) => ({ x: p.x, y: p.y }))
   288→  let i = 1
   289→  while (i + 1 < out.length) {
   290→    const d1x = Math.sign(out[i].x - out[i - 1].x)
   291→    const d1y = Math.sign(out[i].y - out[i - 1].y)
   292→    const d2x = Math.sign(out[i + 1].x - out[i].x)
   293→    const d2y = Math.sign(out[i + 1].y - out[i].y)
   294→    const opposite =
   295→      (d1x !== 0 && d2x !== 0 && d1x === -d2x) || (d1y !== 0 && d2y !== 0 && d1y === -d2y)
   296→    if (opposite) {
   297→      // Drop the corner; the two segments collapse into their net run.
   298→      out.splice(i, 1)
   299→      if (i > 1) i-- // the fold may cascade with the run before it
   300→    } else {
   301→      i++
   302→    }
   303→  }
   304→  return simplifyPath(out)
   305→}

   311→/**
   312→ * Builds the classic orthogonal connector between two anchors:
   313→ * stub out of the source, bend toward the target, stub into the target.
   314→ * `midShift` slides the free corridor lane (head-on / same-direction shapes)
   315→ * sideways so parallel edges can occupy distinct lanes.
   316→ */
   317→export function simpleRoute(p0: Pt, d0: Pt, p3: Pt, d3: Pt, stub = STUB, midShift = 0): Pt[] {
   318→  const s0 = { x: p0.x + d0.x * stub, y: p0.y + d0.y * stub }
   319→  const s1 = { x: p3.x + d3.x * stub, y: p3.y + d3.y * stub }
   320→  const pts: Pt[] = [p0, s0]
   321→  const h0 = d0.x !== 0
   322→  const h3 = d3.x !== 0
   323→
   324→  if (h0 && h3) {
   325→    if (d0.x === -d3.x) {
   326→      // Head-on (e.g. right → left): straight lane when aligned, Z-bend else.
   327→      const forward = d0.x > 0 ? s0.x <= s1.x : s0.x >= s1.x
   328→      if (forward) {
   329→        if (Math.abs(s0.y - s1.y) < 0.5) {
   330→          pts.push(s1)
   331→        } else {
   332→          const mx = (s0.x + s1.x) / 2 + midShift
   333→          pts.push({ x: mx, y: s0.y }, { x: mx, y: s1.y }, s1)
   334→        }
   335→      } else {
   336→        const mx = d0.x > 0 ? Math.max(s0.x, s1.x) + stub : Math.min(s0.x, s1.x) - stub
   337→        pts.push({ x: mx, y: s0.y }, { x: mx, y: s1.y }, s1)
   338→      }
   339→    } else {
   340→      // Same direction: swing around the far side (S/U shape).
   341→      const mx = d0.x > 0 ? Math.max(s0.x, s1.x) + stub : Math.min(s0.x, s1.x) - stub
   342→      pts.push({ x: mx, y: s0.y }, { x: mx, y: s1.y }, s1)
   343→    }
   344→  } else if (!h0 && !h3) {
   345→    if (d0.y === -d3.y) {
   346→      const forward = d0.y > 0 ? s0.y <= s1.y : s0.y >= s1.y
   347→      if (forward) {
   348→        if (Math.abs(s0.x - s1.x) < 0.5) {
   349→          pts.push(s1)
   350→        } else {
   351→          const my = (s0.y + s1.y) / 2 + midShift
   352→          pts.push({ x: s0.x, y: my }, { x: s1.x, y: my }, s1)
   353→        }
   354→      } else {
   355→        const my = d0.y > 0 ? Math.max(s0.y, s1.y) + stub : Math.min(s0.y, s1.y) - stub
   356→        pts.push({ x: s0.x, y: my }, { x: s1.x, y: my }, s1)
   357→      }
   358→    } else {
   359→      const my = d0.y > 0 ? Math.max(s0.y, s1.y) + stub : Math.min(s0.y, s1.y) - stub
   360→      pts.push({ x: s0.x, y: my }, { x: s1.x, y: my }, s1)
   361→    }
   362→  } else {
   363→    // Perpendicular stubs: single elbow where they meet.
   364→    pts.push(h0 ? { x: s1.x, y: s0.y } : { x: s0.x, y: s1.y }, s1)
   365→  }
   366→
   367→  pts.push(p3)
   368→  return simplifyPath(pts)
   369→}

   375→interface RankedCombo {
   376→  source: Side4
   377→  target: Side4
   378→  score: number
   379→  /** Naive route is free of obstacle hits. */
   380→  clean: boolean
   381→}
   382→
   383→// Temporal side-choice memory (edge id → last chosen sides). Routing the same
   384→// edge again prefers the previously chosen combination while it still yields
   385→// a clean route, so near-tied combos stop flapping whenever an unrelated node
   386→// moves a fraction of a pixel. Purely a preference — a blocked or clearly
   387→// worse remembered combo is dropped immediately.
   388→const sidesMemory = new Map<string, Sides>()
   389→
   390→/** How much better (relative + absolute) a new combo must be to overcome the
   391→    stability bias of the remembered sides. */
   392→const SWITCH_RATIO = 0.72
   393→const SWITCH_MARGIN = 20
   394→
   395→function pruneSidesMemory(liveEdgeIds: Set<string>) {
   396→  // Only sweep occasionally (amortised): when the map has clearly outgrown
   397→  // the live edge set, drop entries whose edge no longer exists.
   398→  if (sidesMemory.size <= liveEdgeIds.size * 3 + 24) return
   399→  for (const key of sidesMemory.keys()) {
   400→    if (!liveEdgeIds.has(key)) sidesMemory.delete(key)
   401→  }
   402→}
   403→
   404→function rankCombos(srcRect: Rect, tgtRect: Rect, othersInf: Rect[], ownShrunk: Rect[]): RankedCombo[] {
   405→  const sc = rectCenter(srcRect)
   406→  const tc = rectCenter(tgtRect)
   407→  const dxc = tc.x - sc.x
   408→  const dyc = tc.y - sc.y
   409→  const distC = Math.hypot(dxc, dyc) || 1
   410→  const toT = { x: dxc / distC, y: dyc / distC }
   411→  const toS = { x: -toT.x, y: -toT.y }
   412→
   413→  const combos: RankedCombo[] = []
   414→  for (const ss of SIDE_ORDER) {
   415→    for (const ts of SIDE_ORDER) {
   416→      const a0 = anchorPoint(srcRect, ss)
   417→      const a3 = anchorPoint(tgtRect, ts)
   418→      const pts = simpleRoute(a0, OUTWARD[ss], a3, OUTWARD[ts])
   419→      let score = polyLength(pts)
   420→      const clean =
   421→        !polylineHits(pts, othersInf) && !polylineHits(pts.slice(1, -1), ownShrunk)
   422→      if (!clean) score += BLOCKED_PENALTY
   423→      const d0 = OUTWARD[ss]
   424→      const d3 = OUTWARD[ts]
   425→      score += (1 - (d0.x * toT.x + d0.y * toT.y)) * FACING_WEIGHT
   426→      score += (1 - (d3.x * toS.x + d3.y * toS.y)) * FACING_WEIGHT
   427→      if (ss === ts) score += SAME_SIDE_PENALTY
   428→      // Deterministic tie-break; "top" is checked first so an over-the-top
   429→      // detour wins an otherwise tied reroute (matches expectations when a
   430→      // middle node blocks the direct corridor).
   431→      score += (SIDE_ORDER.indexOf(ss) + SIDE_ORDER.indexOf(ts)) * 0.01
   432→      combos.push({ source: ss, target: ts, score, clean })
   433→    }
   434→  }
   435→  combos.sort((a, b) => a.score - b.score)
   436→  return combos
   437→}
   438→
   439→/**
   440→ * Picks the best side combination. When `preferred` (the previously chosen
   441→ * sides for this edge) still routes clean it is kept unless the best clean
   442→ * combo is substantially shorter — this hysteresis keeps edge shapes stable
   443→ * while unrelated nodes move or during the middle of a drag.
   444→ */
   445→export function chooseSides(
   446→  srcRect: Rect,
   447→  tgtRect: Rect,
   448→  othersInf: Rect[],
   449→  ownShrunk: Rect[],
   450→  preferred?: Sides,
   451→): Sides {
   452→  const ranked = rankCombos(srcRect, tgtRect, othersInf, ownShrunk)
   453→  const bestClean = ranked.find((c) => c.clean)
   454→  if (preferred) {
   455→    const prev = ranked.find((c) => c.source === preferred.source && c.target === preferred.target)
   456→    if (prev) {
   457→      if (prev.clean) {
   458→        const clearlyBetter =
   459→          bestClean !== undefined && bestClean.score < prev.score * SWITCH_RATIO - SWITCH_MARGIN
   460→        if (!clearlyBetter) return { source: prev.source, target: prev.target }
   461→      } else if (bestClean) {
   462→        // Remembered shape is now blocked — take the best clean route.
   463→        return { source: bestClean.source, target: bestClean.target }
   464→      } else {
   465→        // Everything is blocked: keep the remembered shape so A* detours
   466→        // around obstacles without flipping the attachment sides.
   467→        return { source: prev.source, target: prev.target }
   468→      }
   469→    }
   470→  }
   471→  const pick = bestClean ?? ranked[0]
   472→  return { source: pick.source, target: pick.target }
   473→}

   491→/**
   492→ * Offset (px, along the border) for this edge's attachment on `nodeId`/`side`.
   493→ * All edges attaching to the same border are sorted by where their far end
   494→ * sits (edge id as tie-break) and spread apart (compressed if the border is
   495→ * too small for the full spacing). Bidirectional pairs therefore get
   496→ * symmetric ±lanes on both ends.
   497→ */
   498→function attachmentOffset(
   499→  selfId: string,
   500→  nodeId: string,
   501→  side: Side4,
   502→  edges: EdgeRef[],
   503→  boxById: Map<string, Box>,
   504→  sidesFor: (e: EdgeRef) => Sides,
   505→): number {
   506→  const box = boxById.get(nodeId)
   507→  if (!box) return 0
   508→  const axis: "x" | "y" = side === "top" || side === "bottom" ? "x" : "y"
   509→  const peers: { id: string; key: number }[] = []
   510→  for (const e of edges) {
   511→    const s = sidesFor(e)
   512→    let role: "source" | "target" | null = null
   513→    if (e.source === nodeId && s.source === side) role = "source"
   514→    else if (e.target === nodeId && s.target === side) role = "target"
   515→    if (!role) continue
   516→    const otherId = role === "source" ? e.target : e.source
   517→    const other = boxById.get(otherId)
   518→    const oc = other ? rectCenter(other.rect) : { x: 0, y: 0 }
   519→    peers.push({ id: e.id, key: axis === "x" ? oc.x : oc.y })
   520→  }
   521→  if (peers.length === 0) return 0
   522→  peers.sort((a, b) => a.key - b.key || (a.id < b.id ? -1 : a.id > b.id ? 1 : 0))
   523→  const idx = peers.findIndex((p) => p.id === selfId)
   524→  if (idx < 0) return 0
   525→  // Keep the anchor on the border segment; compress the ladder when the
   526→  // border cannot fit the full spacing.
   527→  const half = halfFor(box.rect, side)
   528→  const spacing = peers.length > 1 ? Math.min(LANE_SPACING, (2 * half) / (peers.length - 1)) : LANE_SPACING
   529→  const raw = (idx - (peers.length - 1) / 2) * spacing
   530→  return clamp(raw, -half, half)
   531→}

   534→// Anchor de-confliction for parallel final approaches

   537→function intervalsOverlap(a: [number, number], b: [number, number]): boolean {
   538→  return a[0] <= b[1] && b[0] <= a[1]
   539→}
   540→
   541→/** u-range spanned by an edge's corridor between its anchor node and far node. */
   542→function corridorBand(edge: EdgeRef, anchorNodeId: string, side: Side4, boxById: Map<string, Box>): [number, number] {
   543→  const horizontalCorridor = side === "left" || side === "right"
   544→  const farId = edge.source === anchorNodeId ? edge.target : edge.source
   545→  const a = boxById.get(anchorNodeId)
   546→  const b = boxById.get(farId)
   547→  if (!a || !b) return [Infinity, -Infinity]
   548→  const lo = horizontalCorridor ? (r: Rect) => r.x : (r: Rect) => r.y
   549→  const hi = horizontalCorridor ? (r: Rect) => r.x + r.width : (r: Rect) => r.y + r.height
   550→  return [Math.min(lo(a.rect), lo(b.rect)) - ANCHOR_GAP, Math.max(hi(a.rect), hi(b.rect)) + ANCHOR_GAP]
   551→}
   552→
   553→/**
   554→ * Final approach segments into left/right borders run horizontally at the
   555→ * anchor's y (into top/bottom borders: vertically at its x). When two edges
   556→ * anchor at nearly the same coordinate on such borders and their corridors
   557→ * overlap, their approaches would render as one merged line. This slides own
   558→ * anchor along its border to the nearest clear slot. Deterministic from both
   559→ * sides: the edge with more sliding headroom (ties: higher id) is the one
   560→ * that moves, and everyone evaluates conflicts against original offsets.
   561→ */
   562→function deconflictAnchor(
   563→  self: EdgeRef,
   564→  nodeId: string,
   565→  side: Side4,
   566→  off: number,
   567→  edges: EdgeRef[],
   568→  boxById: Map<string, Box>,
   569→  sidesFor: (e: EdgeRef) => Sides,
   570→  offsetsFor: (e: EdgeRef) => { source: number; target: number },
   571→): number {
   572→  const box = boxById.get(nodeId)
   573→  if (!box) return off
   574→  const axis: "x" | "y" = side === "top" || side === "bottom" ? "x" : "y"
   575→  const myV = borderV(box.rect, side, off)
   576→  const myBand = corridorBand(self, nodeId, side, boxById)
   577→  const myHeadroom = halfFor(box.rect, side) - Math.abs(off)
   578→
   579→  const conflicts: { id: string; v: number; headroom: number }[] = []
   580→  for (const peer of edges) {
   581→    if (peer.id === self.id) continue
   582→    const ps = sidesFor(peer)
   583→    const pOffs = offsetsFor(peer)
   584→    const attachments: [string, Side4, number][] = [
   585→      [peer.source, ps.source, pOffs.source],
   586→      [peer.target, ps.target, pOffs.target],
   587→    ]
   588→    for (const [pNode, pSide, pOff] of attachments) {
   589→      const pAxis: "x" | "y" = pSide === "top" || pSide === "bottom" ? "x" : "y"
   590→      if (pAxis !== axis) continue
   591→      const pBox = boxById.get(pNode)
   592→      if (!pBox) continue
   593→      const pv = borderV(pBox.rect, pSide, pOff)
   594→      if (Math.abs(pv - myV) >= ANCHOR_CLEAR) continue
   595→      if (!intervalsOverlap(myBand, corridorBand(peer, pNode, pSide, boxById))) continue
   596→      conflicts.push({ id: peer.id, v: pv, headroom: halfFor(pBox.rect, pSide) - Math.abs(pOff) })
   597→    }
   598→  }
   599→  if (conflicts.length === 0) return off
   600→
   601→  const mustMove = conflicts.some(
   602→    (c) => myHeadroom > c.headroom || (myHeadroom === c.headroom && self.id > c.id),
   603→  )
   604→  if (!mustMove) return off
   605→
   606→  const avoid = conflicts.map((c) => c.v)
   607→  const half = halfFor(box.rect, side)
   608→  for (let step = 1; step <= 6; step++) {
   609→    for (const dir of [1, -1]) {
   610→      const candidate = off + dir * step * ANCHOR_CLEAR
   611→      if (Math.abs(candidate) > half) continue
   612→      const cv = borderV(box.rect, side, candidate)
   613→      if (avoid.every((v) => Math.abs(cv - v) >= ANCHOR_CLEAR)) return candidate
   614→    }
   615→  }
   616→  return off
   617→}
   618→
   619→/**
   620→ * Bidirectional pairs run two head-on corridors over the same span. Each
   621→ * edge slides its corridor lane half a lane-width toward its own target so
   622→ * the two routes become parallel lines instead of interweaving.
   623→ */
   624→function pairLaneShift(
   625→  self: EdgeRef,
   626→  sides: Sides,
   627→  edges: EdgeRef[],
   628→  boxById: Map<string, Box>,
   629→  sidesFor: (e: EdgeRef) => Sides,
   630→): number {
   631→  const reverse = edges.find((e) => e.id !== self.id && e.source === self.target && e.target === self.source)
   632→  if (!reverse) return 0
   633→  // Only meaningful for head-on combos (opposite sides).
   634→  const d0 = OUTWARD[sides.source]
   635→  const d3 = OUTWARD[sides.target]
   636→  if (d0.x !== -d3.x || d0.y !== -d3.y) return 0
   637→  const rs = sidesFor(reverse)
   638→  if (rs.source !== sides.target || rs.target !== sides.source) return 0
   639→  const sc = rectCenter(boxById.get(self.source)!.rect)
   640→  const tc = rectCenter(boxById.get(self.target)!.rect)
   641→  const horizontal = sides.source === "left" || sides.source === "right"
   642→  const dir = horizontal ? Math.sign(tc.x - sc.x) : Math.sign(tc.y - sc.y)
   643→  return (LANE_SPACING / 2) * dir
   644→}

   647→// Grid A* fallback for fully blocked corridors

   650→/**
   651→ * Orthogonal A* on a coarse grid. Obstacles must already be inflated. Returns
   652→ * interior waypoints from `s0` to `s1` (excluding the endpoints), or null.
   653→ *
   654→ * When `exitDir` is given the first step is forced along that direction (the
   655→ * source stub axis) and when `entryDir` is given the final step into the goal
   656→ * must travel along it (the direction into the target stub) — without these
   657→ * constraints the grid path used to hook/double-back right at the anchors.
   658→ */
   659→function astarRoute(s0: Pt, s1: Pt, obstacles: Rect[], exitDir?: Pt, entryDir?: Pt): Pt[] | null {
   660→  if (exitDir && entryDir) {
   661→    const constrained = astarSearch(s0, s1, obstacles, exitDir, entryDir)
   662→    if (constrained) return constrained
   663→  }
   664→  return astarSearch(s0, s1, obstacles)
   665→}
   666→
   667→function astarSearch(s0: Pt, s1: Pt, obstacles: Rect[], exitDir?: Pt, entryDir?: Pt): Pt[] | null {
   668→  const cellBase = 16
   669→  let minX = Math.min(s0.x, s1.x)
   670→  let minY = Math.min(s0.y, s1.y)
   671→  let maxX = Math.max(s0.x, s1.x)
   672→  let maxY = Math.max(s0.y, s1.y)
   673→  for (const o of obstacles) {
   674→    minX = Math.min(minX, o.x)
   675→    minY = Math.min(minY, o.y)
   676→    maxX = Math.max(maxX, o.x + o.width)
   677→    maxY = Math.max(maxY, o.y + o.height)
   678→  }
   679→  const margin = cellBase * 3
   680→  minX -= margin
   681→  minY -= margin
   682→  maxX += margin
   683→  maxY += margin
   684→
   685→  let cell = cellBase
   686→  let cols = Math.ceil((maxX - minX) / cell) + 1
   687→  let rows = Math.ceil((maxY - minY) / cell) + 1
   688→  while (cols * rows > 60000) {
   689→    cell *= 2
   690→    cols = Math.ceil((maxX - minX) / cell) + 1
   691→    rows = Math.ceil((maxY - minY) / cell) + 1
   692→  }
   693→
   694→  // Grid line coordinates. The exact stub-axis coordinates of both endpoints
   695→  // are inserted as grid lines, so the constrained exit/entry steps run along
   696→  // the stub axes with zero jog (the previous uniform grid forced the anchor
   697→  // alignment to shift points, which rendered as diagonal kinks).
   698→  const xs: number[] = []
   699→  for (let c = 0; c < cols; c++) xs.push(minX + c * cell)
   700→  const ys: number[] = []
   701→  for (let r = 0; r < rows; r++) ys.push(minY + r * cell)
   702→  const ensureLine = (arr: number[], v: number) => {
   703→    for (let i = 0; i < arr.length; i++) {
   704→      if (Math.abs(arr[i] - v) <= 0.01) return
   705→    }
   706→    let best = 0
   707→    for (let i = 1; i < arr.length; i++) {
   708→      if (Math.abs(arr[i] - v) < Math.abs(arr[best] - v)) best = i
   709→    }
   710→    arr.splice(arr[best] < v ? best + 1 : best, 0, v)
   711→  }
   712→  ensureLine(xs, s0.x)
   713→  ensureLine(xs, s1.x)
   714→  ensureLine(ys, s0.y)
   715→  ensureLine(ys, s1.y)
   716→  // An inserted endpoint line can sit right next to the uniform line it
   717→  // replaced — drop the near-duplicate (endpoint lines always win).
   718→  const squeeze = (arr: number[], exact: number[]) => {
   719→    const isExact = (v: number) => exact.some((k) => Math.abs(k - v) <= 0.01)
   720→    const out: number[] = []
   721→    for (const v of arr) {
   722→      const prev = out[out.length - 1]
   723→      if (prev !== undefined && v - prev < 2) {
   724→        if (isExact(prev)) continue // keep the exact line, drop the near-dup
   725→        if (isExact(v)) out.pop() // replace the uniform line with the exact one
   726→        else continue // two near-duplicate uniform lines — keep the first
   727→      }
   728→      out.push(v)
   729→    }
   730→    return out
   731→  }
   732→  const sx = squeeze(xs, [s0.x, s1.x])
   733→  const sy = squeeze(ys, [s0.y, s1.y])
   734→  xs.length = 0
   735→  xs.push(...sx)
   736→  ys.length = 0
   737→  ys.push(...sy)
   738→  cols = xs.length
   739→  rows = ys.length
   740→
   741→  const size = cols * rows
   742→  const blocked = new Uint8Array(size)
   743→  const cellIdx = (c: number, r: number) => r * cols + c
   744→  for (let r = 0; r < rows; r++) {
   745→    const y = ys[r]
   746→    const hh = (r + 1 < rows ? ys[r + 1] : ys[r] + cell) - y
   747→    for (let c = 0; c < cols; c++) {
   748→      const x = xs[c]
   749→      const ww = (c + 1 < cols ? xs[c + 1] : xs[c] + cell) - x
   750→      for (const o of obstacles) {
   751→        if (x < o.x + o.width + 2 && x + ww > o.x - 2 && y < o.y + o.height + 2 && y + hh > o.y - 2) {
   752→          blocked[cellIdx(c, r)] = 1
   753→          break
   754→        }
   755→      }
   756→    }
   757→  }
   758→
   759→  const nearestIndex = (arr: number[], v: number) => {
   760→    let lo = 0
   761→    let hi = arr.length - 1
   762→    while (lo < hi) {
   763→      const mid = (lo + hi) >> 1
   764→      if (arr[mid] < v) lo = mid + 1
   765→      else hi = mid
   766→    }
   767→    if (lo > 0 && Math.abs(arr[lo - 1] - v) < Math.abs(arr[lo] - v)) lo--
   768→    return lo
   769→  }
   770→  const clearNear = (p: Pt) => {
   771→    const c0 = nearestIndex(xs, p.x)
   772→    const r0 = nearestIndex(ys, p.y)
   773→    for (let dr = -1; dr <= 1; dr++) {
   774→      for (let dc = -1; dc <= 1; dc++) {
   775→        const c = c0 + dc
   776→        const r = r0 + dr
   777→        if (c >= 0 && c < cols && r >= 0 && r < rows) blocked[cellIdx(c, r)] = 0
   778→      }
   779→    }
   780→  }
   781→  clearNear(s0)
   782→  clearNear(s1)
   783→
   784→  const startC = nearestIndex(xs, s0.x)
   785→  const startR = nearestIndex(ys, s0.y)
   786→  const goalC = nearestIndex(xs, s1.x)
   787→  const goalR = nearestIndex(ys, s1.y)
   788→  const startIdx = cellIdx(startC, startR)
   789→  const goalIdx = cellIdx(goalC, goalR)
   790→
   791→  // State = cell × direction (0:+x, 1:-x, 2:+y, 3:-y).
   792→  const DIRS: Pt[] = [
   793→    { x: 1, y: 0 },
   794→    { x: -1, y: 0 },
   795→    { x: 0, y: 1 },
   796→    { x: 0, y: -1 },
   797→  ]
   798→  // Direction the route must leave the source stub along, and the direction
   799→  // it must travel when arriving at the target stub (-1 = unconstrained).
   800→  const seedDir = exitDir ? DIRS.findIndex((d) => d.x === exitDir.x && d.y === exitDir.y) : -1
   801→  const wantDir = entryDir ? DIRS.findIndex((d) => d.x === entryDir.x && d.y === entryDir.y) : -1
   802→  const TURN_COST = 0.4
   803→  const states = size * 4
   804→  const gScore = new Float64Array(states).fill(Infinity)
   805→  const parent = new Int32Array(states).fill(-1)
   806→  const closed = new Uint8Array(states)
   807→
   808→  const h = (c: number, r: number) => Math.abs(c - goalC) + Math.abs(r - goalR)
   809→
   810→  // Minimal binary heap over (f, state).
   811→  const heapF: number[] = []
   812→  const heapS: number[] = []
   813→  const push = (f: number, s: number) => {
   814→    heapF.push(f)
   815→    heapS.push(s)
   816→    let i = heapF.length - 1
   817→    while (i > 0) {
   818→      const p = (i - 1) >> 1
   819→      if (heapF[p] <= heapF[i]) break
   820→      ;[heapF[p], heapF[i]] = [heapF[i], heapF[p]]
   821→      ;[heapS[p], heapS[i]] = [heapS[i], heapS[p]]
   822→      i = p
   823→    }
   824→  }
   825→  const pop = (): number => {
   826→    const top = heapS[0]
   827→    const lf = heapF.pop() as number
   828→    const ls = heapS.pop() as number
   829→    if (heapF.length > 0) {
   830→      heapF[0] = lf
   831→      heapS[0] = ls
   832→      let i = 0
   833→      for (;;) {
   834→        const l = 2 * i + 1
   835→        const r = l + 1
   836→        let m = i
   837→        if (l < heapF.length && heapF[l] < heapF[m]) m = l
   838→        if (r < heapF.length && heapF[r] < heapF[m]) m = r
   839→        if (m === i) break
   840→        ;[heapF[m], heapF[i]] = [heapF[i], heapF[m]]
   841→        ;[heapS[m], heapS[i]] = [heapS[i], heapS[m]]
   842→        i = m
   843→      }
   844→    }
   845→    return top
   846→  }
   847→
   848→  for (let d = 0; d < 4; d++) {
   849→    if (seedDir >= 0 && d !== seedDir) continue
   850→    gScore[startIdx * 4 + d] = 0
   851→    push(h(startC, startR), startIdx * 4 + d)
   852→  }
   853→
   854→  let goalState = -1
   855→  while (heapF.length > 0) {
   856→    const state = pop()
   857→    if (closed[state]) continue
   858→    closed[state] = 1
   859→    const cellId = Math.floor(state / 4)
   860→    const dir = state % 4
   861→    if (cellId === goalIdx) {
   862→      // Only accept an arrival that continues straight into the target stub;
   863→      // other approach angles would bend the last segment into a hook.
   864→      if (wantDir >= 0 && dir !== wantDir) continue
   865→      goalState = state
   866→      break
   867→    }
   868→    const c = cellId % cols
   869→    const r = Math.floor(cellId / cols)
   870→    const g = gScore[state]
   871→    for (let nd = 0; nd < 4; nd++) {
   872→      const dc = DIRS[nd].x
   873→      const dr = DIRS[nd].y
   874→      const nc = c + dc
   875→      const nr = r + dr
   876→      if (nc < 0 || nc >= cols || nr < 0 || nr >= rows) continue
   877→      const nCell = cellIdx(nc, nr)
   878→      if (blocked[nCell]) continue
   879→      const ng = g + 1 + (nd === dir ? 0 : TURN_COST)
   880→      const nState = nCell * 4 + nd
   881→      if (ng < gScore[nState] - 1e-9) {
   882→        gScore[nState] = ng
   883→        parent[nState] = state
   884→        push(ng + h(nc, nr), nState)
   885→      }
   886→    }
   887→  }
   888→
   889→  if (goalState < 0) return null
   890→
   891→  const rev: Pt[] = []
   892→  let cur = goalState
   893→  while (cur >= 0) {
   894→    const cellId = Math.floor(cur / 4)
   895→    const c = cellId % cols
   896→    const r = Math.floor(cellId / cols)
   897→    rev.push({ x: xs[c], y: ys[r] })
   898→    cur = parent[cur]
   899→  }
   900→  rev.reverse()
   901→  // With the endpoint-snapped grid the start cell is exactly (s0.x, s0.y), the
   902→  // goal cell exactly (s1.x, s1.y), and the constrained first/last steps run
   903→  // along the stub axes — no anchor alignment (and thus no jog) is needed.
   904→  const pts = rev.slice(1, -1)
   905→  return pruneBacktracks(simplifyPath(pts))
   906→}

   922→/**
   923→ * Computes the smart route for one edge:
   924→ * sides → lane offsets → path (fast naive route, A* detour as fallback).
   925→ */
   926→export function routeEdge(input: RouteInput): RouteResult | null {
   927→  const { edgeId, sourceId, targetId, boxes, edges } = input
   928→  const boxById = new Map<string, Box>()
   929→  for (const b of boxes) boxById.set(b.id, b)
   930→  const srcBox = boxById.get(sourceId)
   931→  const tgtBox = boxById.get(targetId)
   932→  if (!srcBox || !tgtBox) return null
   933→
   934→  const sidesCache = new Map<string, Sides>()
   935→  const sidesFor = (e: EdgeRef): Sides => {
   936→    const hit = sidesCache.get(e.id)
   937→    if (hit) return hit
   938→    const a = boxById.get(e.source)
   939→    const b = boxById.get(e.target)
   940→    let result: Sides
   941→    if (!a || !b || (a.rect.width <= 0 || a.rect.height <= 0 || b.rect.width <= 0 || b.rect.height <= 0)) {
   942→      result = { source: "right", target: "left" }
   943→    } else {
   944→      // Container ancestors of the endpoints are never obstacles for their
   945→      // edges (a child edge must cross the container border).
   946→      const excluded = obstacleExclusionsFor(e.source, e.target, boxes, boxById)
   947→      const othersInf = boxes
   948→        .filter((bx) => !excluded.has(bx.id))
   949→        .map((bx) => inflateRect(bx.rect, OBSTACLE_PAD))
   950→      const ownShrunk = [shrinkRect(a.rect, 1), shrinkRect(b.rect, 1)]
   951→      result = chooseSides(a.rect, b.rect, othersInf, ownShrunk, sidesMemory.get(e.id))
   952→    }
   953→    sidesCache.set(e.id, result)
   954→    sidesMemory.set(e.id, result)
   955→    return result
   956→  }
   957→
   958→  const offsCache = new Map<string, { source: number; target: number }>()
   959→  const offsetsFor = (e: EdgeRef): { source: number; target: number } => {
   960→    const hit = offsCache.get(e.id)
   961→    if (hit) return hit
   962→    const s = sidesFor(e)
   963→    const result = {
   964→      source: attachmentOffset(e.id, e.source, s.source, edges, boxById, sidesFor),
   965→      target: attachmentOffset(e.id, e.target, s.target, edges, boxById, sidesFor),
   966→    }
   967→    offsCache.set(e.id, result)
   968→    return result
   969→  }
   970→
   971→  const self: EdgeRef = { id: edgeId, source: sourceId, target: targetId }
   972→  // Drop memory entries for edges that no longer exist (amortised sweep).
   973→  pruneSidesMemory(new Set(edges.map((e) => e.id)))
   974→  const sides = sidesFor(self)
   975→
   976→  // Spread on shared borders.
   977→  const baseOffs = offsetsFor(self)
   978→  let srcOff = baseOffs.source
   979→  let tgtOff = baseOffs.target
   980→
   981→  // Mirrored *perpendicular* pairs (one edge L-shaped one way, its reverse
   982→  // the other way) interweave unless their anchor order at one shared border
   983→  // is reversed. The border to fix is the leader edge's (smaller id) target
   984→  // border — both edges compute the same leader, so they agree.
   985→  const reverse = edges.find((e) => e.id !== edgeId && e.source === targetId && e.target === sourceId)
   986→  if (reverse) {
   987→    const rs = sidesFor(reverse)
   988→    const mirrored = rs.source === sides.target && rs.target === sides.source
   989→    const d0 = OUTWARD[sides.source]
   990→    const d3 = OUTWARD[sides.target]
   991→    const headOn = d0.x === -d3.x && d0.y === -d3.y
   992→    const perpendicular = d0.x !== 0 !== (d3.x !== 0)
   993→    if (mirrored && perpendicular) {
   994→      const leader = self.id < reverse.id ? self : reverse
   995→      const rOffs = offsetsFor(reverse)
   996→      if (leader.id === self.id) {
   997→        // Reverse the order at my target border (partner's source border).
   998→        const sigma = -Math.sign(srcOff - rOffs.target || (self.id < reverse.id ? 1 : -1))
   999→        const desired = sigma > 0 ? Math.max(tgtOff, rOffs.source) : Math.min(tgtOff, rOffs.source)
  1000→        if (desired !== tgtOff) tgtOff = desired
  1001→      } else {
  1002→        // Leader's target border is my source border: I take the slot that
  1003→        // keeps the two borders' orders opposed (leader picks the other one).
  1004→        const sigma = -Math.sign(rOffs.source - tgtOff || (reverse.id < self.id ? 1 : -1))
  1005→        const desired = sigma > 0 ? Math.min(srcOff, rOffs.target) : Math.max(srcOff, rOffs.target)
  1006→        if (desired !== srcOff) srcOff = desired
  1007→      }
  1008→    }
  1009→  }
  1010→
  1011→  // Slide anchors whose parallel approaches would collide with another
  1012→  // edge's approach.
  1013→  srcOff = deconflictAnchor(self, sourceId, sides.source, srcOff, edges, boxById, sidesFor, offsetsFor)
  1014→  tgtOff = deconflictAnchor(self, targetId, sides.target, tgtOff, edges, boxById, sidesFor, offsetsFor)
  1015→
  1016→  // Corridor lane: counter-shift by the endpoint spread plus the bidirectional
  1017→  // pair bias so parallel edges keep a constant gap along the whole run.
  1018→  const midShift = (srcOff - tgtOff) / 2 + pairLaneShift(self, sides, edges, boxById, sidesFor)
  1019→
  1020→  // Same containment rule for this edge's own obstacle set.
  1021→  const selfExcluded = obstacleExclusionsFor(sourceId, targetId, boxes, boxById)
  1022→  const othersInf = boxes
  1023→    .filter((bx) => !selfExcluded.has(bx.id))
  1024→    .map((bx) => inflateRect(bx.rect, OBSTACLE_PAD))
  1025→  const ownShrunk = [shrinkRect(srcBox.rect, 1), shrinkRect(tgtBox.rect, 1)]
  1026→
  1027→  const isClean = (pts: Pt[]) => !polylineHits(pts, othersInf) && !polylineHits(pts.slice(1, -1), ownShrunk)
  1028→
  1029→  const buildSimple = (ss: Side4, ts: Side4, shift: number): Pt[] => {
  1030→    const a0 = anchorPoint(srcBox.rect, ss, ss === sides.source ? srcOff : 0)
  1031→    const a3 = anchorPoint(tgtBox.rect, ts, ts === sides.target ? tgtOff : 0)
  1032→    return pruneBacktracks(simpleRoute(a0, OUTWARD[ss], a3, OUTWARD[ts], STUB, shift))
  1033→  }
  1034→
  1035→  // 1) Fast path: first clean combo already avoids everything. The chosen
  1036→  //    (hysteresis-stable) combo is tried first so unrelated movement cannot
  1037→  //    flip the shape to a marginally shorter alternative. If the full lane
  1038→  //    shift pushes the corridor into an obstacle, degrade it gracefully
  1039→  //    (half, then zero) — never flip its sign, which would re-introduce
  1040→  //    pair interweaving.
  1041→  const ranked = rankCombos(srcBox.rect, tgtBox.rect, othersInf, ownShrunk)
  1042→  const chosen = ranked.find((c) => c.source === sides.source && c.target === sides.target)
  1043→  const ordered = chosen ? [chosen, ...ranked.filter((c) => c !== chosen)] : ranked
  1044→  for (const combo of ordered) {
  1045→    if (!combo.clean) continue
  1046→    for (const shift of [midShift, midShift / 2, 0]) {
  1047→      const pts = buildSimple(combo.source, combo.target, shift)
  1048→      if (isClean(pts)) {
  1049→        return { points: pts, sourceSide: combo.source, targetSide: combo.target }
  1050→      }
  1051→    }
  1052→  }
  1053→
  1054→  // 2) Nothing clean: A* detours for the most promising combos, constrained
  1055→  //    to leave/enter along the stub axes so the detour has no anchor hooks.
  1056→  //    The chosen combo is evaluated first to keep the shape stable.
  1057→  const candidates: RankedCombo[] = []
  1058→  if (chosen) candidates.push(chosen)
  1059→  for (const combo of ranked) {
  1060→    if (candidates.length >= ASTAR_CANDIDATES) break
  1061→    if (!candidates.includes(combo)) candidates.push(combo)
  1062→  }
  1063→  let best: { pts: Pt[]; cost: number; ss: Side4; ts: Side4 } | null = null
  1064→  for (const combo of candidates) {
  1065→    const a0 = anchorPoint(srcBox.rect, combo.source, combo.source === sides.source ? srcOff : 0)
  1066→    const a3 = anchorPoint(tgtBox.rect, combo.target, combo.target === sides.target ? tgtOff : 0)
  1067→    const d0 = OUTWARD[combo.source]
  1068→    const d3 = OUTWARD[combo.target]
  1069→    const s0 = { x: a0.x + d0.x * STUB, y: a0.y + d0.y * STUB }
  1070→    const s1 = { x: a3.x + d3.x * STUB, y: a3.y + d3.y * STUB }
  1071→    const astarObstacles = [...othersInf, ...ownShrunk]
  1072→    const mid = astarRoute(s0, s1, astarObstacles, d0, { x: -d3.x, y: -d3.y })
  1073→    if (!mid) continue
  1074→    const pts = pruneBacktracks(simplifyPath([a0, s0, ...mid, s1, a3]))
  1075→    const cost = polyLength(pts) + countTurns(pts) * 8
  1076→    if (!best || cost < best.cost) best = { pts, cost, ss: combo.source, ts: combo.target }
  1077→  }
  1078→  if (best) return { points: best.pts, sourceSide: best.ss, targetSide: best.ts }
  1079→
  1080→  // 3) Last resort: the chosen combo's naive route even if it grazes
  1081→  //    something (keeps the previous shape instead of jumping elsewhere).
  1082→  const fallback = chosen ?? ranked[0]
  1083→  return {
  1084→    points: buildSimple(fallback.source, fallback.target, midShift),
  1085→    sourceSide: fallback.source,
  1086→    targetSide: fallback.target,
  1087→  }
  1088→}
  1089→
  1090→function countTurns(pts: Pt[]): number {
  1091→  let turns = 0
  1092→  for (let i = 1; i + 1 < pts.length; i++) {
  1093→    const a = pts[i - 1]
  1094→    const b = pts[i]
  1095→    const c = pts[i + 1]
  1096→    if (Math.abs((b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x)) > 0.01) turns++
  1097→  }
  1098→  return turns
  1099→}

  1105→/** Renders a polyline as an SVG path with quadratic-rounded corners. */
  1106→export function roundedPath(points: Pt[], radius = CORNER_RADIUS): string {
  1107→  const f = (v: number) => {
  1108→    const r = Math.round(v * 100) / 100
  1109→    return Number.isFinite(r) ? String(r) : "0"
  1110→  }
  1111→  if (points.length === 0) return ""
  1112→  if (points.length === 1) return `M ${f(points[0].x)} ${f(points[0].y)}`
  1113→  let d = `M ${f(points[0].x)} ${f(points[0].y)}`
  1114→  for (let i = 1; i < points.length - 1; i++) {
  1115→    const prev = points[i - 1]
  1116→    const cur = points[i]
  1117→    const next = points[i + 1]
  1118→    const l1 = Math.hypot(cur.x - prev.x, cur.y - prev.y)
  1119→    const l2 = Math.hypot(next.x - cur.x, next.y - cur.y)
  1120→    if (l1 < 0.01 || l2 < 0.01) continue
  1121→    const r = Math.min(radius, l1 / 2, l2 / 2)
  1122→    const a = { x: cur.x - ((cur.x - prev.x) / l1) * r, y: cur.y - ((cur.y - prev.y) / l1) * r }
  1123→    const b = { x: cur.x + ((next.x - cur.x) / l2) * r, y: cur.y + ((next.y - cur.y) / l2) * r }
  1124→    d += ` L ${f(a.x)} ${f(a.y)} Q ${f(cur.x)} ${f(cur.y)} ${f(b.x)} ${f(b.y)}`
  1125→  }
  1126→  const last = points[points.length - 1]
  1127→  d += ` L ${f(last.x)} ${f(last.y)}`
  1128→  return d
  1129→}
