   103→// Unique id for programmatically created nodes — Date.now() alone collides
   104→// when several elements are created within the same millisecond (e.g. rapid

   108→  nodeSeq += 1

   147→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   148→function pickSize(value: unknown): number | undefined {
   149→  const n = typeof value === "string" ? Number(value) : (value as number)
   150→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   151→}

   153→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   154→// explicit width/height (written by NodeResizer) → measured (actual rendered
   155→// size) → style (size at creation). Resizing never updates node.style, so a
   156→// round trip that kept only style would silently reset resized elements to
   157→// their creation dimensions.
   158→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   159→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   160→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   161→  return { width, height }
   162→}

   164→function sanitizeImportPayload(raw: unknown): {
   165→  nodes: Node<ShapeData>[]
   166→  edges: Edge[]
   167→  viewport?: { x: number; y: number; zoom: number }
   168→} {
   169→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   170→    throw new Error('expected a JSON object with a "nodes" array')
   171→  }
   172→  const obj = raw as Record<string, unknown>
   173→  if (!Array.isArray(obj.nodes)) {
   174→    throw new Error('missing "nodes" array')
   175→  }
   176→  const validTypes = new Set(Object.keys(nodeTypes))
   177→  const stamp = Date.now()
   178→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   179→    const record = (n ?? {}) as Record<string, any>
   180→    // Restore the exact exported size: prefer the resized width/height, then
   181→    // the measured size, then the creation style — and keep style in sync so
   182→    // React Flow (node.width ?? node.style.width) renders the same box.
   183→    const { width, height } = resolveNodeSizeRecord(record)
   184→    return {
   185→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   186→      type: validTypes.has(record.type) ? record.type : "rectangle",
   187→      position: {
   188→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   189→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   190→      },
   191→      data: { ...(record.data ?? {}) },
   192→      style: {
   193→        ...(record.style ?? {}),
   194→        ...(width !== undefined ? { width } : {}),
   195→        ...(height !== undefined ? { height } : {}),
   196→      },
   197→      ...(width !== undefined ? { width } : {}),
   198→      ...(height !== undefined ? { height } : {}),
   199→      // Deliberately no `measured` — it belongs to the exporting session and
   200→      // React Flow re-measures the node on mount.
   201→      selected: false,
   202→      dragging: false,
   203→      resizing: false,
   204→    } as unknown as Node<ShapeData>
   205→  })
   206→  const nodeIds = new Set(nodes.map((n) => n.id))
   207→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   208→    .filter(
   209→      (e) =>
   210→        e &&
   211→        typeof e.source === "string" &&
   212→        typeof e.target === "string" &&
   213→        nodeIds.has(e.source) &&
   214→        nodeIds.has(e.target) &&
   215→        e.source !== e.target,
   216→    )
   217→    .map((e, i) => ({
   218→      ...e,
   219→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   220→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   221→      selected: false,
   222→    })) as Edge[]
   223→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   224→  return { nodes, edges, viewport }
   225→}
