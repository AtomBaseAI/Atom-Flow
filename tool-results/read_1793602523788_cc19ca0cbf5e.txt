   214→// ---- Smart Guides (Photoshop-style alignment helpers) ----------------------
   215→// While an element is dragged, its left/center/right and top/middle/bottom
   216→// anchors are compared against every other element's anchors. Within a small
   217→// (screen-space) threshold the drag is snapped to the closest match and a
   218→// magenta guide line is drawn through the aligned axis, exactly like Smart
   219→// Guides in Photoshop.
   220→
   221→type SnapBox = { x: number; y: number; w: number; h: number }
   222→
   223→type SnapGuides = {
   224→  /** Vertical lines (fixed x, spanning y1..y2 in flow coords) */
   225→  vertical: { x: number; y1: number; y2: number }[]
   226→  /** Horizontal lines (fixed y, spanning x1..x2 in flow coords) */
   227→  horizontal: { y: number; x1: number; x2: number }[]
   228→}
   229→
   230→const EMPTY_SNAP_GUIDES: SnapGuides = { vertical: [], horizontal: [] }
   231→const SNAP_THRESHOLD_PX = 6 // screen-space snap distance
   232→const SNAP_GUIDE_PAD = 24 // flow-space extension of the guide beyond both boxes
   233→
   234→function nodeSnapBox(n: Node<ShapeData>): SnapBox {
   235→  const style = (n.style ?? {}) as { width?: number | string; height?: number | string }
   236→  const w = n.measured?.width ?? (typeof style.width === "number" ? style.width : 0)
   237→  const h = n.measured?.height ?? (typeof style.height === "number" ? style.height : 0)
   238→  return { x: n.position.x, y: n.position.y, w, h }
   239→}
   240→
   241→function unionSnapBoxes(a: SnapBox, b: SnapBox): SnapBox {
   242→  const x = Math.min(a.x, b.x)
   243→  const y = Math.min(a.y, b.y)
   244→  return {
   245→    x,
   246→    y,
   247→    w: Math.max(a.x + a.w, b.x + b.w) - x,
   248→    h: Math.max(a.y + a.h, b.y + b.h) - y,
   249→  }
   250→}

   252→/** Renders the active smart-guide lines inside the viewport coordinate space
   253→    so they pan/zoom with the canvas. Lines stay 1 screen px thick at any zoom. */
   254→function SnapGuidesLayer({ guides }: { guides: SnapGuides }) {
   255→  const zoom = useStore((s) => s.transform[2]) || 1
   256→  if (guides.vertical.length === 0 && guides.horizontal.length === 0) return null
   257→  const t = 1 / zoom
   258→  return (
   259→    <ViewportPortal>
   260→      {guides.vertical.map((g, i) => (
   261→        <div
   262→          key={`v-${i}-${Math.round(g.x * 10)}-${Math.round(g.y1)}`}
   263→          aria-hidden
   264→          className="pointer-events-none absolute left-0 top-0"
   265→          style={{
   266→            transform: `translate(${g.x - t / 2}px, ${g.y1}px)`,
   267→            width: t,
   268→            height: Math.max(g.y2 - g.y1, t),
   269→            background: "var(--af-snap-line)",
   270→            zIndex: 60,
   271→          }}
   272→        />
   273→      ))}
   274→      {guides.horizontal.map((g, i) => (
   275→        <div
   276→          key={`h-${i}-${Math.round(g.y * 10)}-${Math.round(g.x1)}`}
   277→          aria-hidden
   278→          className="pointer-events-none absolute left-0 top-0"
   279→          style={{
   280→            transform: `translate(${g.x1}px, ${g.y - t / 2}px)`,

   283→            background: "var(--af-snap-line)",
   284→            zIndex: 60,
   285→          }}

   312→// ---- Theme-aware default colors -----------------------------------------
   313→// Old boards stored the *default* border/text colors as literal hex values
   314→// ("#a3a3a3" border, "#ffffff" text). Those now map to "no explicit color",
   315→// so they render from the --af-* CSS variables and flip with the theme.
   316→// Any other stored value is treated as a user-picked color and marked custom
   317→// so it stays fixed across theme switches.
   318→const LEGACY_DEFAULT_BORDER = "#a3a3a3"
   319→const LEGACY_DEFAULT_TEXT = "#ffffff"
   320→
   321→function migrateShapeDefaults(raw: Record<string, any> | undefined): Record<string, any> {
   322→  const data = { ...(raw ?? {}) }
   323→  const border = typeof data.borderColor === "string" ? data.borderColor.toLowerCase() : undefined
   324→  if (!data.borderColorCustom && (!border || border === LEGACY_DEFAULT_BORDER)) {
   325→    delete data.borderColor
   326→  } else if (data.borderColor) {
   327→    data.borderColorCustom = true
   328→  }
   329→  const font = typeof data.fontColor === "string" ? data.fontColor.toLowerCase() : undefined
   330→  if (!data.fontColorCustom && (!font || font === LEGACY_DEFAULT_TEXT)) {
   331→    delete data.fontColor
   332→  } else if (data.fontColor) {
   333→    data.fontColorCustom = true
   334→  }
   335→  return data
   336→}
   337→
   338→function migrateEdgeDefaults(e: Record<string, any>): Record<string, any> {
   339→  const next: Record<string, any> = { ...e }
   340→  const style = { ...(e.style ?? {}) }
   341→  if (typeof style.stroke === "string" && style.stroke.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   342→    delete style.stroke
   343→  }
   344→  next.style = style
   345→  const marker = next.markerEnd
   346→  if (marker && typeof marker === "object" && typeof marker.color === "string" && marker.color.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   347→    const markerEnd = { ...marker }
   348→    delete markerEnd.color
   349→    next.markerEnd = markerEnd
   350→  }
   351→  return next
   352→}

   367→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   368→function pickSize(value: unknown): number | undefined {
   369→  const n = typeof value === "string" ? Number(value) : (value as number)
   370→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   371→}

   373→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   374→// explicit width/height (written by NodeResizer) → measured (actual rendered
   375→// size) → style (size at creation). Resizing never updates node.style, so a
   376→// round trip that kept only style would silently reset resized elements to
   377→// their creation dimensions.
   378→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   379→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   380→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   381→  return { width, height }
   382→}

   384→function sanitizeImportPayload(raw: unknown): {
   385→  nodes: Node<ShapeData>[]
   386→  edges: Edge[]
   387→  viewport?: { x: number; y: number; zoom: number }
   388→} {
   389→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   390→    throw new Error('expected a JSON object with a "nodes" array')
   391→  }
   392→  const obj = raw as Record<string, unknown>
   393→  if (!Array.isArray(obj.nodes)) {
   394→    throw new Error('missing "nodes" array')
   395→  }
   396→  const validTypes = new Set(Object.keys(nodeTypes))
   397→  const stamp = Date.now()
   398→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   399→    const record = (n ?? {}) as Record<string, any>
   400→    // Restore the exact exported size: prefer the resized width/height, then
   401→    // the measured size, then the creation style — and keep style in sync so
   402→    // React Flow (node.width ?? node.style.width) renders the same box.
   403→    const { width, height } = resolveNodeSizeRecord(record)
   404→    return {
   405→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   406→      type: validTypes.has(record.type) ? record.type : "rectangle",
   407→      position: {
   408→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   409→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   410→      },
   411→      data: migrateShapeDefaults(record.data),
   412→      style: {
   413→        ...(record.style ?? {}),
   414→        ...(width !== undefined ? { width } : {}),
   415→        ...(height !== undefined ? { height } : {}),
   416→      },
   417→      ...(width !== undefined ? { width } : {}),
   418→      ...(height !== undefined ? { height } : {}),
   419→      // Deliberately no `measured` — it belongs to the exporting session and
   420→      // React Flow re-measures the node on mount.
   421→      selected: false,
   422→      dragging: false,
   423→      resizing: false,
   424→    } as unknown as Node<ShapeData>
   425→  })
   426→  const nodeIds = new Set(nodes.map((n) => n.id))
   427→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   428→    .filter(
   429→      (e) =>
   430→        e &&
   431→        typeof e.source === "string" &&
   432→        typeof e.target === "string" &&
   433→        nodeIds.has(e.source) &&
   434→        nodeIds.has(e.target) &&
   435→        e.source !== e.target,
   436→    )
   437→    .map((e, i) => ({
   438→      ...migrateEdgeDefaults(e),
   439→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   440→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   441→      selected: false,
   442→    })) as Edge[]
   443→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   444→  return { nodes, edges, viewport }
   445→}

   658→        const startResize = changes.some((c) => (c as any).type === "dimensions" && (c as any).resizing === true)

   670→      // The drag's intended (unsnapped) positions arrive in the position

   673→      // The final dragging=false change of a drag is corrected too (drag still
   674→      // flagged active via nodeDragActiveRef) so the element stays snapped.
   675→      let effective = changes

   677→        const dragActive = nodeDragActiveRef.current
   678→        const dragChanges = changes.filter(
   679→          (c): c is NodePositionChange<Node<ShapeData>> =>
   680→            c.type === "position" && !!(c as any).position && ((c as any).dragging === true || (dragActive && (c as any).dragging === false)),
   681→        )

   684→          const zoom = rf.getViewport().zoom || 1
   685→          const threshold = SNAP_THRESHOLD_PX / zoom
   686→          const posById = new Map<string, { x: number; y: number }>()
   687→          for (const c of dragChanges) posById.set(c.id, c.position as { x: number; y: number })
   688→
   689→          const draggedBoxes: SnapBox[] = []
   690→          const targetBoxes: SnapBox[] = []
   691→          for (const n of nodes) {
   692→            const box = nodeSnapBox(n)
   693→            const moved = posById.get(n.id)
   694→            if (moved) draggedBoxes.push({ ...box, x: moved.x, y: moved.y })
   695→            else targetBoxes.push(box)
   696→          }
   697→
   698→          if (draggedBoxes.length > 0 && targetBoxes.length > 0) {
   699→            const dBox = draggedBoxes.reduce(unionSnapBoxes)
   700→            const anchorsX = [dBox.x, dBox.x + dBox.w / 2, dBox.x + dBox.w]
   701→            const anchorsY = [dBox.y, dBox.y + dBox.h / 2, dBox.y + dBox.h]
   702→            let bestX: { delta: number; coord: number; box: SnapBox } | null = null
   703→            let bestY: { delta: number; coord: number; box: SnapBox } | null = null
   704→            for (const t of targetBoxes) {
   705→              const tx = [t.x, t.x + t.w / 2, t.x + t.w]
   706→              const ty = [t.y, t.y + t.h / 2, t.y + t.h]
   707→              for (const a of anchorsX) {
   708→                for (const tc of tx) {
   709→                  const d = tc - a
   710→                  if (Math.abs(d) <= threshold && (!bestX || Math.abs(d) < Math.abs(bestX.delta))) {
   711→                    bestX = { delta: d, coord: tc, box: t }
   712→                  }
   713→                }
   714→              }
   715→              for (const a of anchorsY) {
   716→                for (const tc of ty) {
   717→                  const d = tc - a
   718→                  if (Math.abs(d) <= threshold && (!bestY || Math.abs(d) < Math.abs(bestY.delta))) {
   719→                    bestY = { delta: d, coord: tc, box: t }
   720→                  }
   721→                }
   722→              }
   723→            }
   724→
   725→            const dx = bestX?.delta ?? 0
   726→            const dy = bestY?.delta ?? 0
   727→            if (dx !== 0 || dy !== 0) {
   728→              effective = changes.map((c) => {
   729→                if (c.type === "position" && (c as any).position && ((c as any).dragging === true || (dragActive && (c as any).dragging === false))) {
   730→                  const pc = c as NodePositionChange<Node<ShapeData>>
   731→                  return { ...pc, position: { x: pc.position!.x + dx, y: pc.position!.y + dy } }
   732→                }
   733→                return c
   734→              })
   735→            }
   736→
   737→            // Live drag: show guides. End-of-drag event (dragging=false only):
   738→            // position got snapped above, guides are dropped.
   739→            const snapped = { ...dBox, x: dBox.x + dx, y: dBox.y + dy }
   740→            const vertical = liveDrag && bestX
   741→              ? [
   742→                  {
   743→                    x: bestX.coord,
   744→                    y1: Math.min(snapped.y, bestX.box.y) - SNAP_GUIDE_PAD,
   745→                    y2: Math.max(snapped.y + snapped.h, bestX.box.y + bestX.box.h) + SNAP_GUIDE_PAD,
   746→                  },
   747→                ]
   748→              : []
   749→            const horizontal = liveDrag && bestY
   750→              ? [
   751→                  {
   752→                    y: bestY.coord,
   753→                    x1: Math.min(snapped.x, bestY.box.x) - SNAP_GUIDE_PAD,
   754→                    x2: Math.max(snapped.x + snapped.w, bestY.box.x + bestY.box.w) + SNAP_GUIDE_PAD,
   755→                  },
   756→                ]
   757→              : []
   758→            setSnapGuides((prev) => {
   759→              const nextEmpty = vertical.length === 0 && horizontal.length === 0
   760→              const prevEmpty = prev.vertical.length === 0 && prev.horizontal.length === 0
   761→              if (nextEmpty && prevEmpty) return prev
   762→              return { vertical, horizontal }
   763→            })
   764→          } else {
   765→            setSnapGuides((prev) =>
   766→              prev.vertical.length === 0 && prev.horizontal.length === 0 ? prev : EMPTY_SNAP_GUIDES,
   767→            )
   768→          }
   769→        } else if (changes.some((c) => c.type === "position")) {
   770→          // Drag ended or a programmatic move — drop any guides
   771→          setSnapGuides((prev) =>
   772→            prev.vertical.length === 0 && prev.horizontal.length === 0 ? prev : EMPTY_SNAP_GUIDES,
   773→          )
   774→        }
   775→      }
   776→
   777→      onNodesChange(effective)
   778→    },

   780→  )
