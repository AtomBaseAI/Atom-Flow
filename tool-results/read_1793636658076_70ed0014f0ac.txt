    23→/** Rounded 360×200 canvas every figure draws on. */
    24→function Frame({ label, children }: { label: string; children: React.ReactNode }) {
    25→  return (
    26→    <svg viewBox="0 0 360 200" role="img" aria-label={label} className="block h-auto w-full">
    27→      {children}
    28→    </svg>
    29→  )
    30→}

    32→/** Teal / muted / rose arrowhead marker — must live inside <defs> with a unique id. */
    33→function ArrowHead({ id, tone = "teal" }: { id: string; tone?: "teal" | "muted" | "rose" }) {
    34→  const fill =
    35→    tone === "muted"
    36→      ? "fill-muted-foreground/60"
    37→      : tone === "rose"
    38→        ? "fill-rose-500"
    39→        : "fill-teal-500"
    40→  return (
    41→    <marker
    42→      id={id}
    43→      viewBox="0 0 8 8"
    44→      refX={7}
    45→      refY={4}
    46→      markerWidth={6.5}
    47→      markerHeight={6.5}
    48→      orient="auto-start-reverse"
    49→    >
    50→      <path d="M0 0L8 4L0 8Z" className={fill} />
    51→    </marker>
    52→  )
    53→}

    55→/** A UI node box with optional centered label. */
    56→function Box({
    57→  x,
    58→  y,
    59→  w,
    60→  h,
    61→  label,
    62→  rx = 6,
    63→  className,
    64→}: {
    65→  x: number
    66→  y: number
    67→  w: number
    68→  h: number
    69→  label?: string
    70→  rx?: number
    71→  className?: string
    72→}) {
    73→  return (
    74→    <g>
    75→      <rect x={x} y={y} width={w} height={h} rx={rx} strokeWidth={1.5} className={cn(panelCls, className)} />
    76→      {label !== undefined && (
    77→        <text x={x + w / 2} y={y + h / 2 + 3} fontSize={8.5} textAnchor="middle" className="fill-foreground/80">
    78→          {label}
    79→        </text>
    80→      )}
    81→    </g>
    82→  )
    83→}

    85→/** Teal selection grips on the four corners of a box. */
    86→function Grips({ x, y, w, h }: { x: number; y: number; w: number; h: number }) {
    87→  const s = 5
    88→  const pts: Array<[number, number]> = [
    89→    [x - s / 2, y - s / 2],
    90→    [x + w - s / 2, y - s / 2],
    91→    [x - s / 2, y + h - s / 2],
    92→    [x + w - s / 2, y + h - s / 2],
    93→  ]
    94→  return (
    95→    <g>
    96→      {pts.map(([gx, gy], i) => (
    97→        <rect
    98→          key={i}
    99→          x={gx}
   100→          y={gy}
   101→          width={s}
   102→          height={s}
   103→          rx={1}
   104→          strokeWidth={1.2}
   105→          className="fill-background stroke-teal-500"
   106→        />
   107→      ))}
   108→    </g>
   109→  )
   110→}
