   126→/**
   127→ * Static, true-to-scale preview of how the icon will land on the board:
   128→ *  - icon    → bare 96×96 icon
   129→ *  - labeled → 96×96 icon with bottom text strip (classic)
   130→ *  - frame   → 160×120 app-window frame (title top-left, 2× icon top-right)
   131→ */
   132→function IconPreview({
   133→  src,
   134→  lucideComp,
   135→  label,
   136→  style,
   137→}: {
   138→  src: string
   139→  lucideComp?: React.ComponentType<{ className?: string; style?: React.CSSProperties }>
   140→  label: string
   141→  style: IconStyleId
   142→}) {
   143→  const imgSrc = src.startsWith("lucide:") ? "" : src
   144→  const Glyph = lucideComp
   145→  const borderColor = "var(--af-node-border)"
   146→  const textColor = "var(--af-node-text)"
   147→
   148→  if (style === "frame") {
   149→    // Mirrors IconFrameNode: 160×120 app window, 8px title top-left, 2× square icon top-right
   150→    const titleSize = 8
   151→    const frameIconSize = Math.max(16, titleSize * 2)
   152→    return (
   153→      <div
   154→        className="overflow-hidden rounded-[4px] shadow-sm"
   155→        style={{ width: 160, height: 120, background: "var(--af-frame-bg)", border: `1px solid ${borderColor}` }}
   156→      >
   157→        <div className="flex items-center justify-between gap-1 border-b border-black/10 px-1 py-0.5 dark:border-white/15">
   158→          <span
   159→            className="min-w-0 flex-1 truncate text-left font-semibold leading-[1.2]"
   160→            style={{ color: textColor, fontSize: titleSize }}
   161→          >
   162→            {label}
   163→          </span>
   164→          <span className="flex-none" style={{ width: frameIconSize, height: frameIconSize }} aria-hidden>
   165→            {Glyph ? (
   166→              <Glyph className="h-full w-full" style={{ color: borderColor }} aria-hidden />
   167→            ) : imgSrc ? (
   168→              <img src={imgSrc} alt="" className="h-full w-full object-contain" draggable={false} />
   169→            ) : null}
   170→          </span>
   171→        </div>
   172→        <div />
   173→      </div>
   174→    )
   175→  }
   176→
   177→  if (style === "icon") {
   178→    return (
   179→      <div
   180→        className="flex items-center justify-center rounded-[2px]"
   181→        style={{ width: 96, height: 96, border: `1px solid ${borderColor}` }}
   182→      >
   183→        {Glyph ? (
   184→          <Glyph className="h-[80%] w-[80%]" style={{ color: borderColor }} aria-hidden />
   185→        ) : (
   186→          <img src={imgSrc || "/placeholder.svg"} alt="" className="h-[85%] w-[85%] object-contain" draggable={false} />
   187→        )}
   188→      </div>
   189→    )
   190→  }
   191→
   192→  // labeled — icon with bottom text strip (classic)
   193→  return (
   194→    <div
   195→      className="flex flex-col overflow-hidden rounded-[2px]"
   196→      style={{ width: 96, height: 96, border: `1px solid ${borderColor}` }}
   197→    >
   198→      <div className="flex min-h-0 flex-1 items-center justify-center p-1">
   199→        {Glyph ? (
   200→          <Glyph className="h-[85%] w-[85%]" style={{ color: borderColor }} aria-hidden />
   201→        ) : (
   202→          <img src={imgSrc || "/placeholder.svg"} alt="" className="h-[85%] w-[85%] object-contain" draggable={false} />
   203→        )}
   204→      </div>
   205→      <div className="flex min-h-[24px] flex-none items-center justify-center border-t border-black/10 px-1 dark:border-white/15">
   206→        <span className="w-full truncate text-center text-[8px] font-bold leading-tight" style={{ color: textColor }}>
   207→          {label}
   208→        </span>
   209→      </div>
   210→    </div>
   211→  )
   212→}
