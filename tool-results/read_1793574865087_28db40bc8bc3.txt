   211→// ---- Smart Guides (Photoshop-style alignment helpers) ----------------------
   212→// While an element is dragged, its left/center/right and top/middle/bottom
   213→// anchors are compared against every other element's anchors. Within a small
   214→// (screen-space) threshold the drag is snapped to the closest match and a
   215→// magenta guide line is drawn through the aligned axis, exactly like Smart
   216→// Guides in Photoshop.
   217→
   218→type SnapBox = { x: number; y: number; w: number; h: number }
   219→
   220→type SnapGuides = {
   221→  /** Vertical lines (fixed x, spanning y1..y2 in flow coords) */
   222→  vertical: { x: number; y1: number; y2: number }[]
   223→  /** Horizontal lines (fixed y, spanning x1..x2 in flow coords) */
   224→  horizontal: { y: number; x1: number; x2: number }[]
   225→}
   226→
   227→const EMPTY_SNAP_GUIDES: SnapGuides = { vertical: [], horizontal: [] }
   228→const SNAP_THRESHOLD_PX = 6 // screen-space snap distance
   229→const SNAP_GUIDE_PAD = 24 // flow-space extension of the guide beyond both boxes
   230→
   231→function nodeSnapBox(n: Node<ShapeData>): SnapBox {
   232→  const style = (n.style ?? {}) as { width?: number | string; height?: number | string }
   233→  const w = n.measured?.width ?? (typeof style.width === "number" ? style.width : 0)
   234→  const h = n.measured?.height ?? (typeof style.height === "number" ? style.height : 0)
   235→  return { x: n.position.x, y: n.position.y, w, h }
   236→}
   237→
   238→function unionSnapBoxes(a: SnapBox, b: SnapBox): SnapBox {
   239→  const x = Math.min(a.x, b.x)
   240→  const y = Math.min(a.y, b.y)
   241→  return {
   242→    x,
   243→    y,
   244→    w: Math.max(a.x + a.w, b.x + b.w) - x,
   245→    h: Math.max(a.y + a.h, b.y + b.h) - y,
   246→  }
   247→}

   249→/** Renders the active smart-guide lines inside the viewport coordinate space

   251→function SnapGuidesLayer({ guides }: { guides: SnapGuides }) {
   252→  const zoom = useStore((s) => s.transform[2]) || 1
   253→  if (guides.vertical.length === 0 && guides.horizontal.length === 0) return null

   260→          aria-hidden
   261→          className="pointer-events-none absolute left-0 top-0"
   262→          style={{
   263→            transform: `translate(${g.x - t / 2}px, ${g.y1}px)`,
   264→            width: t,
   265→            height: Math.max(g.y2 - g.y1, t),
   266→            background: "var(--af-snap-line)",
   267→            zIndex: 60,
   268→          }}
   269→        />
   270→      ))}

   309→// ---- Theme-aware default colors -----------------------------------------
   310→// Old boards stored the *default* border/text colors as literal hex values
   311→// ("#a3a3a3" border, "#ffffff" text). Those now map to "no explicit color",
   312→// so they render from the --af-* CSS variables and flip with the theme.
   313→// Any other stored value is treated as a user-picked color and marked custom
   314→// so it stays fixed across theme switches.
   315→const LEGACY_DEFAULT_BORDER = "#a3a3a3"
   316→const LEGACY_DEFAULT_TEXT = "#ffffff"
   317→
   318→function migrateShapeDefaults(raw: Record<string, any> | undefined): Record<string, any> {
   319→  const data = { ...(raw ?? {}) }
   320→  const border = typeof data.borderColor === "string" ? data.borderColor.toLowerCase() : undefined
   321→  if (!data.borderColorCustom && (!border || border === LEGACY_DEFAULT_BORDER)) {
   322→    delete data.borderColor
   323→  } else if (data.borderColor) {
   324→    data.borderColorCustom = true
   325→  }
   326→  const font = typeof data.fontColor === "string" ? data.fontColor.toLowerCase() : undefined
   327→  if (!data.fontColorCustom && (!font || font === LEGACY_DEFAULT_TEXT)) {
   328→    delete data.fontColor
   329→  } else if (data.fontColor) {
   330→    data.fontColorCustom = true
   331→  }
   332→  return data
   333→}
   334→
   335→function migrateEdgeDefaults(e: Record<string, any>): Record<string, any> {
   336→  const next: Record<string, any> = { ...e }
   337→  const style = { ...(e.style ?? {}) }
   338→  if (typeof style.stroke === "string" && style.stroke.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   339→    delete style.stroke
   340→  }
   341→  next.style = style
   342→  const marker = next.markerEnd
   343→  if (marker && typeof marker === "object" && typeof marker.color === "string" && marker.color.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   344→    const markerEnd = { ...marker }
   345→    delete markerEnd.color
   346→    next.markerEnd = markerEnd
   347→  }
   348→  return next
   349→}

   363→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   364→function pickSize(value: unknown): number | undefined {
   365→  const n = typeof value === "string" ? Number(value) : (value as number)
   366→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   367→}

   369→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   370→// explicit width/height (written by NodeResizer) → measured (actual rendered
   371→// size) → style (size at creation). Resizing never updates node.style, so a
   372→// round trip that kept only style would silently reset resized elements to
   373→// their creation dimensions.
   374→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   375→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   376→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   377→  return { width, height }
   378→}

   380→function sanitizeImportPayload(raw: unknown): {
   381→  nodes: Node<ShapeData>[]
   382→  edges: Edge[]
   383→  viewport?: { x: number; y: number; zoom: number }
   384→} {
   385→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   386→    throw new Error('expected a JSON object with a "nodes" array')
   387→  }
   388→  const obj = raw as Record<string, unknown>
   389→  if (!Array.isArray(obj.nodes)) {
   390→    throw new Error('missing "nodes" array')
   391→  }
   392→  const validTypes = new Set(Object.keys(nodeTypes))
   393→  const stamp = Date.now()
   394→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   395→    const record = (n ?? {}) as Record<string, any>
   396→    // Restore the exact exported size: prefer the resized width/height, then
   397→    // the measured size, then the creation style — and keep style in sync so
   398→    // React Flow (node.width ?? node.style.width) renders the same box.
   399→    const { width, height } = resolveNodeSizeRecord(record)
   400→    return {
   401→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   402→      type: validTypes.has(record.type) ? record.type : "rectangle",
   403→      position: {
   404→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   405→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   406→      },
   407→      data: migrateShapeDefaults(record.data),
   408→      style: {
   409→        ...(record.style ?? {}),
   410→        ...(width !== undefined ? { width } : {}),
   411→        ...(height !== undefined ? { height } : {}),
   412→      },
   413→      ...(width !== undefined ? { width } : {}),
   414→      ...(height !== undefined ? { height } : {}),
   415→      // Deliberately no `measured` — it belongs to the exporting session and
   416→      // React Flow re-measures the node on mount.
   417→      selected: false,
   418→      dragging: false,
   419→      resizing: false,
   420→    } as unknown as Node<ShapeData>
   421→  })
   422→  const nodeIds = new Set(nodes.map((n) => n.id))
   423→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   424→    .filter(
   425→      (e) =>
   426→        e &&
   427→        typeof e.source === "string" &&
   428→        typeof e.target === "string" &&
   429→        nodeIds.has(e.source) &&
   430→        nodeIds.has(e.target) &&
   431→        e.source !== e.target,
   432→    )
   433→    .map((e, i) => ({
   434→      ...migrateEdgeDefaults(e),
   435→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   436→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   437→      selected: false,
   438→    })) as Edge[]
   439→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   440→  return { nodes, edges, viewport }
   441→}

   691→            else targetBoxes.push(box)

   695→            const dBox = draggedBoxes.reduce(unionSnapBoxes)

   697→            const anchorsY = [dBox.y, dBox.y + dBox.h / 2, dBox.y + dBox.h]

   699→            let bestY: { delta: number; coord: number; box: SnapBox } | null = null

   707→                    bestX = { delta: d, coord: tc, box: t }

   711→              for (const a of anchorsY) {
   712→                for (const tc of ty) {

   715→                    bestY = { delta: d, coord: tc, box: t }
   716→                  }
   717→                }
   718→              }
   719→            }

   721→            const dx = bestX?.delta ?? 0
   722→            const dy = bestY?.delta ?? 0

   725→                if (c.type === "position" && (c as any).position && ((c as any).dragging === true || (dragActive && (c as any).dragging === false))) {

   729→                return c
   730→              })

   751→                  },
   752→                ]
   753→              : []

   756→              const prevEmpty = prev.vertical.length === 0 && prev.horizontal.length === 0

   762→              prev.vertical.length === 0 && prev.horizontal.length === 0 ? prev : EMPTY_SNAP_GUIDES,
   763→            )
   764→          }

   768→            prev.vertical.length === 0 && prev.horizontal.length === 0 ? prev : EMPTY_SNAP_GUIDES,
   769→          )
   770→        }
   771→      }
   772→
   773→      onNodesChange(effective)
   774→    },
