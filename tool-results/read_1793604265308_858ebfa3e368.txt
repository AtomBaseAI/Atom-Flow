   235→// ---- Smart Guides (Photoshop-style alignment helpers) ----------------------
   236→// While an element is dragged, its left/center/right and top/middle/bottom
   237→// anchors are compared against every other element's anchors. Within a small
   238→// (screen-space) threshold the drag is snapped to the closest match and a
   239→// magenta guide line is drawn through the aligned axis, exactly like Smart
   240→// Guides in Photoshop.
   241→
   242→type SnapBox = { x: number; y: number; w: number; h: number }
   243→
   244→type SnapGuides = {
   245→  /** Vertical lines (fixed x, spanning y1..y2 in flow coords) */
   246→  vertical: { x: number; y1: number; y2: number }[]
   247→  /** Horizontal lines (fixed y, spanning x1..x2 in flow coords) */
   248→  horizontal: { y: number; x1: number; x2: number }[]
   249→}
   250→
   251→const EMPTY_SNAP_GUIDES: SnapGuides = { vertical: [], horizontal: [] }
   252→const SNAP_THRESHOLD_PX = 6 // screen-space snap distance
   253→const SNAP_GUIDE_PAD = 24 // flow-space extension of the guide beyond both boxes
   254→
   255→function nodeSnapBox(n: Node<ShapeData>): SnapBox {
   256→  const style = (n.style ?? {}) as { width?: number | string; height?: number | string }
   257→  const w = n.measured?.width ?? (typeof style.width === "number" ? style.width : 0)
   258→  const h = n.measured?.height ?? (typeof style.height === "number" ? style.height : 0)
   259→  return { x: n.position.x, y: n.position.y, w, h }
   260→}
   261→
   262→function unionSnapBoxes(a: SnapBox, b: SnapBox): SnapBox {
   263→  const x = Math.min(a.x, b.x)
   264→  const y = Math.min(a.y, b.y)
   265→  return {
   266→    x,
   267→    y,
   268→    w: Math.max(a.x + a.w, b.x + b.w) - x,
   269→    h: Math.max(a.y + a.h, b.y + b.h) - y,
   270→  }
   271→}

   333→// ---- Theme-aware default colors -----------------------------------------
   334→// Old boards stored the *default* border/text colors as literal hex values
   335→// ("#a3a3a3" border, "#ffffff" text). Those now map to "no explicit color",
   336→// so they render from the --af-* CSS variables and flip with the theme.
   337→// Any other stored value is treated as a user-picked color and marked custom
   338→// so it stays fixed across theme switches.
   339→const LEGACY_DEFAULT_BORDER = "#a3a3a3"
   340→const LEGACY_DEFAULT_TEXT = "#ffffff"
   341→
   342→function migrateShapeDefaults(raw: Record<string, any> | undefined): Record<string, any> {
   343→  const data = { ...(raw ?? {}) }
   344→  const border = typeof data.borderColor === "string" ? data.borderColor.toLowerCase() : undefined
   345→  if (!data.borderColorCustom && (!border || border === LEGACY_DEFAULT_BORDER)) {
   346→    delete data.borderColor
   347→  } else if (data.borderColor) {
   348→    data.borderColorCustom = true
   349→  }
   350→  const font = typeof data.fontColor === "string" ? data.fontColor.toLowerCase() : undefined
   351→  if (!data.fontColorCustom && (!font || font === LEGACY_DEFAULT_TEXT)) {
   352→    delete data.fontColor
   353→  } else if (data.fontColor) {
   354→    data.fontColorCustom = true
   355→  }
   356→  return data
   357→}
   358→
   359→function migrateEdgeDefaults(e: Record<string, any>): Record<string, any> {
   360→  const next: Record<string, any> = { ...e }
   361→  const style = { ...(e.style ?? {}) }
   362→  if (typeof style.stroke === "string" && style.stroke.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   363→    delete style.stroke
   364→  }
   365→  next.style = style
   366→  const marker = next.markerEnd
   367→  if (marker && typeof marker === "object" && typeof marker.color === "string" && marker.color.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   368→    const markerEnd = { ...marker }
   369→    delete markerEnd.color
   370→    next.markerEnd = markerEnd
   371→  }
   372→  return next
   373→}

   390→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   391→function pickSize(value: unknown): number | undefined {
   392→  const n = typeof value === "string" ? Number(value) : (value as number)
   393→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   394→}

   396→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   397→// explicit width/height (written by NodeResizer) → measured (actual rendered
   398→// size) → style (size at creation). Resizing never updates node.style, so a
   399→// round trip that kept only style would silently reset resized elements to
   400→// their creation dimensions.
   401→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   402→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   403→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   404→  return { width, height }
   405→}

   407→function sanitizeImportPayload(raw: unknown): {
   408→  nodes: Node<ShapeData>[]
   409→  edges: Edge[]
   410→  viewport?: { x: number; y: number; zoom: number }
   411→} {
   412→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   413→    throw new Error('expected a JSON object with a "nodes" array')
   414→  }
   415→  const obj = raw as Record<string, unknown>
   416→  if (!Array.isArray(obj.nodes)) {
   417→    throw new Error('missing "nodes" array')
   418→  }
   419→  const validTypes = new Set(Object.keys(nodeTypes))
   420→  const stamp = Date.now()
   421→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   422→    const record = (n ?? {}) as Record<string, any>
   423→    // Restore the exact exported size: prefer the resized width/height, then
   424→    // the measured size, then the creation style — and keep style in sync so
   425→    // React Flow (node.width ?? node.style.width) renders the same box.
   426→    const { width, height } = resolveNodeSizeRecord(record)
   427→    return {
   428→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   429→      type: validTypes.has(record.type) ? record.type : "rectangle",
   430→      position: {
   431→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   432→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   433→      },
   434→      data: migrateShapeDefaults(record.data),
   435→      style: {
   436→        ...(record.style ?? {}),
   437→        ...(width !== undefined ? { width } : {}),
   438→        ...(height !== undefined ? { height } : {}),
   439→      },
   440→      ...(width !== undefined ? { width } : {}),
   441→      ...(height !== undefined ? { height } : {}),
   442→      // Deliberately no `measured` — it belongs to the exporting session and
   443→      // React Flow re-measures the node on mount.
   444→      selected: false,
   445→      dragging: false,
   446→      resizing: false,
   447→    } as unknown as Node<ShapeData>
   448→  })
   449→  const nodeIds = new Set(nodes.map((n) => n.id))
   450→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   451→    .filter(
   452→      (e) =>
   453→        e &&
   454→        typeof e.source === "string" &&
   455→        typeof e.target === "string" &&
   456→        nodeIds.has(e.source) &&
   457→        nodeIds.has(e.target) &&
   458→        e.source !== e.target,
   459→    )
   460→    .map((e, i) => ({
   461→      ...migrateEdgeDefaults(e),
   462→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   463→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   464→      selected: false,
   465→    })) as Edge[]
   466→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   467→  return { nodes, edges, viewport }
   468→}

   717→            if (moved) draggedBoxes.push({ ...box, x: moved.x, y: moved.y })
   718→            else targetBoxes.push(box)
   719→          }
   720→
   721→          if (draggedBoxes.length > 0 && targetBoxes.length > 0) {
   722→            const dBox = draggedBoxes.reduce(unionSnapBoxes)
   723→            const anchorsX = [dBox.x, dBox.x + dBox.w / 2, dBox.x + dBox.w]
   724→            const anchorsY = [dBox.y, dBox.y + dBox.h / 2, dBox.y + dBox.h]
   725→            let bestX: { delta: number; coord: number; box: SnapBox } | null = null
   726→            let bestY: { delta: number; coord: number; box: SnapBox } | null = null
   727→            for (const t of targetBoxes) {
   728→              const tx = [t.x, t.x + t.w / 2, t.x + t.w]
   729→              const ty = [t.y, t.y + t.h / 2, t.y + t.h]
   730→              for (const a of anchorsX) {
   731→                for (const tc of tx) {
   732→                  const d = tc - a
   733→                  if (Math.abs(d) <= threshold && (!bestX || Math.abs(d) < Math.abs(bestX.delta))) {
   734→                    bestX = { delta: d, coord: tc, box: t }
   735→                  }
   736→                }
   737→              }
   738→              for (const a of anchorsY) {
   739→                for (const tc of ty) {
   740→                  const d = tc - a
   741→                  if (Math.abs(d) <= threshold && (!bestY || Math.abs(d) < Math.abs(bestY.delta))) {
   742→                    bestY = { delta: d, coord: tc, box: t }
   743→                  }
   744→                }
   745→              }
   746→            }
   747→
   748→            const dx = bestX?.delta ?? 0
   749→            const dy = bestY?.delta ?? 0
   750→            if (dx !== 0 || dy !== 0) {
   751→              effective = changes.map((c) => {
   752→                if (c.type === "position" && (c as any).position && ((c as any).dragging === true || (dragActive && (c as any).dragging === false))) {
   753→                  const pc = c as NodePositionChange<Node<ShapeData>>
   754→                  return { ...pc, position: { x: pc.position!.x + dx, y: pc.position!.y + dy } }
   755→                }
   756→                return c
   757→              })
   758→            }
   759→
   760→            // Live drag: show guides. End-of-drag event (dragging=false only):
   761→            // position got snapped above, guides are dropped.
   762→            const snapped = { ...dBox, x: dBox.x + dx, y: dBox.y + dy }

   769→                  },

   771→              : []
   772→            const horizontal = liveDrag && bestY
   773→              ? [
   774→                  {
   775→                    y: bestY.coord,
   776→                    x1: Math.min(snapped.x, bestY.box.x) - SNAP_GUIDE_PAD,
   777→                    x2: Math.max(snapped.x + snapped.w, bestY.box.x + bestY.box.w) + SNAP_GUIDE_PAD,
   778→                  },
   779→                ]
   780→              : []
   781→            setSnapGuides((prev) => {
   782→              const nextEmpty = vertical.length === 0 && horizontal.length === 0
   783→              const prevEmpty = prev.vertical.length === 0 && prev.horizontal.length === 0
   784→              if (nextEmpty && prevEmpty) return prev
   785→              return { vertical, horizontal }
   786→            })
   787→          } else {
   788→            setSnapGuides((prev) =>
   789→              prev.vertical.length === 0 && prev.horizontal.length === 0 ? prev : EMPTY_SNAP_GUIDES,
   790→            )
   791→          }
   792→        } else if (changes.some((c) => c.type === "position")) {
   793→          // Drag ended or a programmatic move — drop any guides
   794→          setSnapGuides((prev) =>
   795→            prev.vertical.length === 0 && prev.horizontal.length === 0 ? prev : EMPTY_SNAP_GUIDES,
   796→          )
   797→        }
   798→      }
   799→
   800→      onNodesChange(effective)
   801→    },
