   130→/**
   131→ * Static, true-to-scale preview of how the icon will land on the board:
   132→ *  - icon    → bare 96×96 icon
   133→ *  - labeled → 96×96 icon with bottom text strip (classic)
   134→ *  - frame   → 160×120 app-window frame (title top-left, 2× icon top-right)
   135→ */
   136→function IconPreview({
   137→  src,
   138→  lucideComp,
   139→  label,
   140→  style,
   141→}: {
   142→  src: string
   143→  lucideComp?: React.ComponentType<{ className?: string; style?: React.CSSProperties }>
   144→  label: string
   145→  style: IconStyleId
   146→}) {
   147→  const imgSrc = src.startsWith("lucide:") ? "" : src
   148→  const Glyph = lucideComp
   149→  const borderColor = "var(--af-node-border)"
   150→  const textColor = "var(--af-node-text)"
   151→
   152→  if (style === "frame") {
   153→    // Mirrors IconFrameNode: 160×120 app window, 8px title top-left, 2× square icon top-right
   154→    const titleSize = 8
   155→    const frameIconSize = Math.max(16, titleSize * 2)
   156→    return (
   157→      <div
   158→        className="overflow-hidden rounded-[4px] shadow-sm"
   159→        style={{ width: 160, height: 120, background: "var(--af-frame-bg)", border: `1px solid ${borderColor}` }}
   160→      >
   161→        <div className="flex items-center justify-between gap-1 border-b border-black/10 px-1 py-0.5 dark:border-white/15">
   162→          <span
   163→            className="min-w-0 flex-1 truncate text-left font-semibold leading-[1.2]"
   164→            style={{ color: textColor, fontSize: titleSize }}
   165→          >
   166→            {label}
   167→          </span>
   168→          <span className="flex-none" style={{ width: frameIconSize, height: frameIconSize }} aria-hidden>
   169→            {Glyph ? (
   170→              <Glyph className="h-full w-full" style={{ color: borderColor }} aria-hidden />
   171→            ) : imgSrc ? (
   172→              <img src={imgSrc} alt="" className="h-full w-full object-contain" draggable={false} />
   173→            ) : null}
   174→          </span>
   175→        </div>
   176→        <div />
   177→      </div>
   178→    )
   179→  }
   180→
   181→  if (style === "icon") {
   182→    return (
   183→      <div
   184→        className="flex items-center justify-center rounded-[2px]"
   185→        style={{ width: 96, height: 96, border: `1px solid ${borderColor}` }}
   186→      >
   187→        {Glyph ? (
   188→          <Glyph className="h-[80%] w-[80%]" style={{ color: borderColor }} aria-hidden />
   189→        ) : (
   190→          <img src={imgSrc || "/placeholder.svg"} alt="" className="h-[85%] w-[85%] object-contain" draggable={false} />
   191→        )}
   192→      </div>
   193→    )
   194→  }
   195→
   196→  // labeled — icon with bottom text strip (classic)
   197→  return (
   198→    <div
   199→      className="flex flex-col overflow-hidden rounded-[2px]"
   200→      style={{ width: 96, height: 96, border: `1px solid ${borderColor}` }}
   201→    >
   202→      <div className="flex min-h-0 flex-1 items-center justify-center p-1">
   203→        {Glyph ? (
   204→          <Glyph className="h-[85%] w-[85%]" style={{ color: borderColor }} aria-hidden />
   205→        ) : (
   206→          <img src={imgSrc || "/placeholder.svg"} alt="" className="h-[85%] w-[85%] object-contain" draggable={false} />
   207→        )}
   208→      </div>
   209→      <div className="flex min-h-[24px] flex-none items-center justify-center border-t border-black/10 px-1 dark:border-white/15">
   210→        <span className="w-full truncate text-center text-[8px] font-bold leading-tight" style={{ color: textColor }}>
   211→          {label}
   212→        </span>
   213→      </div>
   214→    </div>
   215→  )
   216→}
