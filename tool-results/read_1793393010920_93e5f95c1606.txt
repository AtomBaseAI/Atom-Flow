   186→   186→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   187→   187→function pickSize(value: unknown): number | undefined {
   188→   188→  const n = typeof value === "string" ? Number(value) : (value as number)
   189→   189→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   190→   190→}
   191→   191→
   192→   192→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   193→   193→// explicit width/height (written by NodeResizer) → measured (actual rendered
   194→   194→// size) → style (size at creation). Resizing never updates node.style, so a
   195→   195→// round trip that kept only style would silently reset resized elements to
   196→   196→// their creation dimensions.
   197→   197→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   198→   198→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   199→   199→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   200→   200→  return { width, height }
   201→   201→}

   203→   203→function sanitizeImportPayload(raw: unknown): {
   204→   204→  nodes: Node<ShapeData>[]
   205→   205→  edges: Edge[]
   206→   206→  viewport?: { x: number; y: number; zoom: number }
   207→   207→} {
   208→   208→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   209→   209→    throw new Error('expected a JSON object with a "nodes" array')
   210→   210→  }
   211→   211→  const obj = raw as Record<string, unknown>
   212→   212→  if (!Array.isArray(obj.nodes)) {
   213→   213→    throw new Error('missing "nodes" array')
   214→   214→  }
   215→   215→  const validTypes = new Set(Object.keys(nodeTypes))
   216→   216→  const stamp = Date.now()
   217→   217→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   218→   218→    const record = (n ?? {}) as Record<string, any>
   219→   219→    // Restore the exact exported size: prefer the resized width/height, then
   220→   220→    // the measured size, then the creation style — and keep style in sync so
   221→   221→    // React Flow (node.width ?? node.style.width) renders the same box.
   222→   222→    const { width, height } = resolveNodeSizeRecord(record)
   223→   223→    return {
   224→   224→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   225→   225→      type: validTypes.has(record.type) ? record.type : "rectangle",
   226→   226→      position: {
   227→   227→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   228→   228→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   229→   229→      },
   230→   230→      data: { ...(record.data ?? {}) },
   231→   231→      style: {
   232→   232→        ...(record.style ?? {}),
   233→   233→        ...(width !== undefined ? { width } : {}),
   234→   234→        ...(height !== undefined ? { height } : {}),
   235→   235→      },
   236→   236→      ...(width !== undefined ? { width } : {}),
   237→   237→      ...(height !== undefined ? { height } : {}),
   238→   238→      // Deliberately no `measured` — it belongs to the exporting session and
   239→   239→      // React Flow re-measures the node on mount.
   240→   240→      selected: false,
   241→   241→      dragging: false,
   242→   242→      resizing: false,
   243→   243→    } as unknown as Node<ShapeData>
   244→   244→  })
   245→   245→  const nodeIds = new Set(nodes.map((n) => n.id))
   246→   246→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   247→   247→    .filter(
   248→   248→      (e) =>
   249→   249→        e &&
   250→   250→        typeof e.source === "string" &&
   251→   251→        typeof e.target === "string" &&
   252→   252→        nodeIds.has(e.source) &&
   253→   253→        nodeIds.has(e.target) &&
   254→   254→        e.source !== e.target,
   255→   255→    )
   256→   256→    .map((e, i) => ({
   257→   257→      ...e,
   258→   258→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   259→   259→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   260→   260→      selected: false,
   261→   261→    })) as Edge[]
   262→   262→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   263→   263→  return { nodes, edges, viewport }
   264→   264→}
