    50→function ToolBtn({
    51→  active,
    52→  onClick,
    53→  label,
    54→  children,
    55→}: {
    56→  active?: boolean
    57→  onClick?: () => void
    58→  label: string
    59→  children: React.ReactNode
    60→}) {
    61→  return (
    62→    <Tooltip>
    63→      <TooltipTrigger asChild>
    64→        <button
    65→          aria-label={label}
    66→          onClick={onClick}
    67→          className={cn(
    68→            "flex h-9 w-9 items-center justify-center rounded-md border border-border bg-card text-foreground",
    69→            "hover:bg-accent focus:outline-none focus:ring-2 focus:ring-teal-500",
    70→            active && "border-teal-500 text-teal-600 dark:text-white",
    71→          )}
    72→        >
    73→          {children}
    74→        </button>
    75→      </TooltipTrigger>
    76→      <TooltipContent side="right" align="center" sideOffset={6}>
    77→        {label}
    78→      </TooltipContent>
    79→    </Tooltip>
    80→  )
    81→}

    83→function LucideTile({
    84→  entry,
    85→  selected,
    86→  onSelect,
    87→}: {
    88→  entry: LucideEntry
    89→  selected: boolean
    90→  onSelect: () => void
    91→}) {
    92→  const IconComp = entry.comp
    93→  return (
    94→    <button
    95→      onClick={onSelect}
    96→      className={cn(
    97→        "flex aspect-square items-center justify-center rounded border p-1.5 transition-colors",
    98→        selected
    99→          ? "border-teal-600 bg-teal-500/10 text-teal-700 dark:text-white ring-1 ring-teal-600"
   100→          : "border-border bg-muted text-foreground hover:bg-accent",
   101→      )}
   102→      title={entry.name + (entry.experimental ? " (experimental)" : "")}
   103→      aria-pressed={selected}

   105→    >
   106→      <IconComp className="h-5 w-5" strokeWidth={1.75} aria-hidden />
   107→    </button>
   108→  )
   109→}
