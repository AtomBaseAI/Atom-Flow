   109→// Unique id for programmatically created nodes — Date.now() alone collides
   110→// when several elements are created within the same millisecond (e.g. rapid
   111→// double-adds), which corrupts the node lookup.

   113→function newNodeId(): string {
   114→  nodeSeq += 1
   115→  return `n-${Date.now().toString(36)}-${nodeSeq.toString(36)}-${Math.random().toString(36).slice(2, 6)}`

   121→// View-only lock mode (persists across reloads): hides the editing chrome,
   122→// forces the hand tool and freezes node/edge interactions while panning and

   126→// The lock flag lives in localStorage (an external system), so it is exposed
   127→// through useSyncExternalStore: no setState-in-effect, and the server snapshot
   128→// (unlocked) prevents hydration mismatches while still re-rendering locked
   129→// right after hydration when the stored flag is "1".
   130→const lockListeners = new Set<() => void>()

   132→function subscribeLock(onChange: () => void) {
   133→  lockListeners.add(onChange)
   134→  return () => {
   135→    lockListeners.delete(onChange)
   136→  }
   137→}

   180→// ---- Theme-aware default colors -----------------------------------------
   181→// Old boards stored the *default* border/text colors as literal hex values
   182→// ("#a3a3a3" border, "#ffffff" text). Those now map to "no explicit color",
   183→// so they render from the --af-* CSS variables and flip with the theme.
   184→// Any other stored value is treated as a user-picked color and marked custom
   185→// so it stays fixed across theme switches.
   186→const LEGACY_DEFAULT_BORDER = "#a3a3a3"
   187→const LEGACY_DEFAULT_TEXT = "#ffffff"
   188→
   189→function migrateShapeDefaults(raw: Record<string, any> | undefined): Record<string, any> {
   190→  const data = { ...(raw ?? {}) }
   191→  const border = typeof data.borderColor === "string" ? data.borderColor.toLowerCase() : undefined
   192→  if (!data.borderColorCustom && (!border || border === LEGACY_DEFAULT_BORDER)) {
   193→    delete data.borderColor
   194→  } else if (data.borderColor) {
   195→    data.borderColorCustom = true
   196→  }
   197→  const font = typeof data.fontColor === "string" ? data.fontColor.toLowerCase() : undefined
   198→  if (!data.fontColorCustom && (!font || font === LEGACY_DEFAULT_TEXT)) {
   199→    delete data.fontColor
   200→  } else if (data.fontColor) {
   201→    data.fontColorCustom = true
   202→  }
   203→  return data
   204→}
   205→
   206→function migrateEdgeDefaults(e: Record<string, any>): Record<string, any> {
   207→  const next: Record<string, any> = { ...e }
   208→  const style = { ...(e.style ?? {}) }
   209→  if (typeof style.stroke === "string" && style.stroke.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   210→    delete style.stroke
   211→  }
   212→  next.style = style
   213→  const marker = next.markerEnd
   214→  if (marker && typeof marker === "object" && typeof marker.color === "string" && marker.color.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   215→    const markerEnd = { ...marker }
   216→    delete markerEnd.color
   217→    next.markerEnd = markerEnd
   218→  }
   219→  return next
   220→}

   234→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   235→function pickSize(value: unknown): number | undefined {
   236→  const n = typeof value === "string" ? Number(value) : (value as number)
   237→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   238→}
   239→
   240→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   241→// explicit width/height (written by NodeResizer) → measured (actual rendered
   242→// size) → style (size at creation). Resizing never updates node.style, so a
   243→// round trip that kept only style would silently reset resized elements to
   244→// their creation dimensions.
   245→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   246→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   247→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   248→  return { width, height }
   249→}

   251→function sanitizeImportPayload(raw: unknown): {
   252→  nodes: Node<ShapeData>[]
   253→  edges: Edge[]
   254→  viewport?: { x: number; y: number; zoom: number }
   255→} {
   256→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   257→    throw new Error('expected a JSON object with a "nodes" array')
   258→  }
   259→  const obj = raw as Record<string, unknown>
   260→  if (!Array.isArray(obj.nodes)) {
   261→    throw new Error('missing "nodes" array')
   262→  }
   263→  const validTypes = new Set(Object.keys(nodeTypes))
   264→  const stamp = Date.now()
   265→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   266→    const record = (n ?? {}) as Record<string, any>
   267→    // Restore the exact exported size: prefer the resized width/height, then
   268→    // the measured size, then the creation style — and keep style in sync so
   269→    // React Flow (node.width ?? node.style.width) renders the same box.
   270→    const { width, height } = resolveNodeSizeRecord(record)
   271→    return {
   272→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   273→      type: validTypes.has(record.type) ? record.type : "rectangle",
   274→      position: {
   275→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   276→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   277→      },
   278→      data: migrateShapeDefaults(record.data),
   279→      style: {
   280→        ...(record.style ?? {}),
   281→        ...(width !== undefined ? { width } : {}),
   282→        ...(height !== undefined ? { height } : {}),
   283→      },
   284→      ...(width !== undefined ? { width } : {}),
   285→      ...(height !== undefined ? { height } : {}),
   286→      // Deliberately no `measured` — it belongs to the exporting session and
   287→      // React Flow re-measures the node on mount.
   288→      selected: false,
   289→      dragging: false,
   290→      resizing: false,
   291→    } as unknown as Node<ShapeData>
   292→  })
   293→  const nodeIds = new Set(nodes.map((n) => n.id))
   294→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   295→    .filter(
   296→      (e) =>
   297→        e &&
   298→        typeof e.source === "string" &&
   299→        typeof e.target === "string" &&
   300→        nodeIds.has(e.source) &&
   301→        nodeIds.has(e.target) &&
   302→        e.source !== e.target,
   303→    )
   304→    .map((e, i) => ({
   305→      ...migrateEdgeDefaults(e),
   306→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   307→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   308→      selected: false,
   309→    })) as Edge[]
   310→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   311→  return { nodes, edges, viewport }
   312→}
