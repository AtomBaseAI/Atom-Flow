   209→// ---- Smart Guides (Photoshop-style alignment helpers) ----------------------
   210→// While an element is dragged, its left/center/right and top/middle/bottom
   211→// anchors are compared against every other element's anchors. Within a small
   212→// (screen-space) threshold the drag is snapped to the closest match and a
   213→// magenta guide line is drawn through the aligned axis, exactly like Smart
   214→// Guides in Photoshop.
   215→
   216→type SnapBox = { x: number; y: number; w: number; h: number }
   217→
   218→type SnapGuides = {
   219→  /** Vertical lines (fixed x, spanning y1..y2 in flow coords) */
   220→  vertical: { x: number; y1: number; y2: number }[]
   221→  /** Horizontal lines (fixed y, spanning x1..x2 in flow coords) */
   222→  horizontal: { y: number; x1: number; x2: number }[]
   223→}
   224→
   225→const EMPTY_SNAP_GUIDES: SnapGuides = { vertical: [], horizontal: [] }
   226→const SNAP_THRESHOLD_PX = 6 // screen-space snap distance
   227→const SNAP_GUIDE_PAD = 24 // flow-space extension of the guide beyond both boxes
   228→
   229→function nodeSnapBox(n: Node<ShapeData>): SnapBox {
   230→  const style = (n.style ?? {}) as { width?: number | string; height?: number | string }
   231→  const w = n.measured?.width ?? (typeof style.width === "number" ? style.width : 0)
   232→  const h = n.measured?.height ?? (typeof style.height === "number" ? style.height : 0)
   233→  return { x: n.position.x, y: n.position.y, w, h }
   234→}
   235→
   236→function unionSnapBoxes(a: SnapBox, b: SnapBox): SnapBox {
   237→  const x = Math.min(a.x, b.x)
   238→  const y = Math.min(a.y, b.y)
   239→  return {
   240→    x,
   241→    y,
   242→    w: Math.max(a.x + a.w, b.x + b.w) - x,
   243→    h: Math.max(a.y + a.h, b.y + b.h) - y,
   244→  }
   245→}

   247→/** Renders the active smart-guide lines inside the viewport coordinate space
   248→    so they pan/zoom with the canvas. Lines stay 1 screen px thick at any zoom. */
   249→function SnapGuidesLayer({ guides }: { guides: SnapGuides }) {
   250→  const zoom = useStore((s) => s.transform[2]) || 1
   251→  if (guides.vertical.length === 0 && guides.horizontal.length === 0) return null
   252→  const t = 1 / zoom

   254→    <ViewportPortal>

   257→          key={`v-${i}-${Math.round(g.x * 10)}-${Math.round(g.y1)}`}
   258→          aria-hidden
   259→          className="pointer-events-none absolute left-0 top-0"
   260→          style={{
   261→            transform: `translate(${g.x - t / 2}px, ${g.y1}px)`,
   262→            width: t,
   263→            height: Math.max(g.y2 - g.y1, t),
   264→            background: "var(--af-snap-line)",
   265→            zIndex: 60,
   266→          }}
   267→        />
   268→      ))}
   269→      {guides.horizontal.map((g, i) => (
   270→        <div
   271→          key={`h-${i}-${Math.round(g.y * 10)}-${Math.round(g.x1)}`}
   272→          aria-hidden
   273→          className="pointer-events-none absolute left-0 top-0"

   275→            transform: `translate(${g.x1}px, ${g.y - t / 2}px)`,

   278→            background: "var(--af-snap-line)",

   307→// ---- Theme-aware default colors -----------------------------------------
   308→// Old boards stored the *default* border/text colors as literal hex values
   309→// ("#a3a3a3" border, "#ffffff" text). Those now map to "no explicit color",
   310→// so they render from the --af-* CSS variables and flip with the theme.
   311→// Any other stored value is treated as a user-picked color and marked custom
   312→// so it stays fixed across theme switches.
   313→const LEGACY_DEFAULT_BORDER = "#a3a3a3"
   314→const LEGACY_DEFAULT_TEXT = "#ffffff"
   315→
   316→function migrateShapeDefaults(raw: Record<string, any> | undefined): Record<string, any> {
   317→  const data = { ...(raw ?? {}) }
   318→  const border = typeof data.borderColor === "string" ? data.borderColor.toLowerCase() : undefined
   319→  if (!data.borderColorCustom && (!border || border === LEGACY_DEFAULT_BORDER)) {
   320→    delete data.borderColor
   321→  } else if (data.borderColor) {
   322→    data.borderColorCustom = true
   323→  }
   324→  const font = typeof data.fontColor === "string" ? data.fontColor.toLowerCase() : undefined
   325→  if (!data.fontColorCustom && (!font || font === LEGACY_DEFAULT_TEXT)) {
   326→    delete data.fontColor
   327→  } else if (data.fontColor) {
   328→    data.fontColorCustom = true
   329→  }
   330→  return data
   331→}
   332→
   333→function migrateEdgeDefaults(e: Record<string, any>): Record<string, any> {
   334→  const next: Record<string, any> = { ...e }
   335→  const style = { ...(e.style ?? {}) }
   336→  if (typeof style.stroke === "string" && style.stroke.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   337→    delete style.stroke
   338→  }
   339→  next.style = style
   340→  const marker = next.markerEnd
   341→  if (marker && typeof marker === "object" && typeof marker.color === "string" && marker.color.toLowerCase() === LEGACY_DEFAULT_BORDER) {
   342→    const markerEnd = { ...marker }
   343→    delete markerEnd.color
   344→    next.markerEnd = markerEnd
   345→  }
   346→  return next
   347→}

   361→// Accepts a finite number (or numeric string) and returns it, otherwise undefined.
   362→function pickSize(value: unknown): number | undefined {
   363→  const n = typeof value === "string" ? Number(value) : (value as number)
   364→  return typeof n === "number" && Number.isFinite(n) ? n : undefined
   365→}

   367→// Resolves a node's CURRENT on-canvas size with a fallback chain:
   368→// explicit width/height (written by NodeResizer) → measured (actual rendered
   369→// size) → style (size at creation). Resizing never updates node.style, so a
   370→// round trip that kept only style would silently reset resized elements to
   371→// their creation dimensions.
   372→function resolveNodeSizeRecord(record: Record<string, any>): { width?: number; height?: number } {
   373→  const width = pickSize(record.width) ?? pickSize(record.measured?.width) ?? pickSize(record.style?.width)
   374→  const height = pickSize(record.height) ?? pickSize(record.measured?.height) ?? pickSize(record.style?.height)
   375→  return { width, height }
   376→}

   378→function sanitizeImportPayload(raw: unknown): {
   379→  nodes: Node<ShapeData>[]
   380→  edges: Edge[]
   381→  viewport?: { x: number; y: number; zoom: number }
   382→} {
   383→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   384→    throw new Error('expected a JSON object with a "nodes" array')
   385→  }
   386→  const obj = raw as Record<string, unknown>
   387→  if (!Array.isArray(obj.nodes)) {
   388→    throw new Error('missing "nodes" array')
   389→  }
   390→  const validTypes = new Set(Object.keys(nodeTypes))
   391→  const stamp = Date.now()
   392→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   393→    const record = (n ?? {}) as Record<string, any>
   394→    // Restore the exact exported size: prefer the resized width/height, then
   395→    // the measured size, then the creation style — and keep style in sync so
   396→    // React Flow (node.width ?? node.style.width) renders the same box.
   397→    const { width, height } = resolveNodeSizeRecord(record)
   398→    return {
   399→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   400→      type: validTypes.has(record.type) ? record.type : "rectangle",
   401→      position: {
   402→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   403→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   404→      },
   405→      data: migrateShapeDefaults(record.data),
   406→      style: {
   407→        ...(record.style ?? {}),
   408→        ...(width !== undefined ? { width } : {}),
   409→        ...(height !== undefined ? { height } : {}),
   410→      },
   411→      ...(width !== undefined ? { width } : {}),
   412→      ...(height !== undefined ? { height } : {}),
   413→      // Deliberately no `measured` — it belongs to the exporting session and
   414→      // React Flow re-measures the node on mount.
   415→      selected: false,
   416→      dragging: false,
   417→      resizing: false,
   418→    } as unknown as Node<ShapeData>
   419→  })
   420→  const nodeIds = new Set(nodes.map((n) => n.id))
   421→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   422→    .filter(
   423→      (e) =>
   424→        e &&
   425→        typeof e.source === "string" &&
   426→        typeof e.target === "string" &&
   427→        nodeIds.has(e.source) &&
   428→        nodeIds.has(e.target) &&
   429→        e.source !== e.target,
   430→    )
   431→    .map((e, i) => ({
   432→      ...migrateEdgeDefaults(e),
   433→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   434→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   435→      selected: false,
   436→    })) as Edge[]
   437→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   438→  return { nodes, edges, viewport }
   439→}

   762→          }

   766→            prev.vertical.length === 0 && prev.horizontal.length === 0 ? prev : EMPTY_SNAP_GUIDES,
   767→          )
   768→        }
   769→      }

   772→    },
