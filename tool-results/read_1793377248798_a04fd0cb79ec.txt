   114→function sanitizeImportPayload(raw: unknown): {
   115→  nodes: Node<ShapeData>[]
   116→  edges: Edge[]
   117→  viewport?: { x: number; y: number; zoom: number }
   118→} {
   119→  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
   120→    throw new Error('expected a JSON object with a "nodes" array')
   121→  }
   122→  const obj = raw as Record<string, unknown>
   123→  if (!Array.isArray(obj.nodes)) {
   124→    throw new Error('missing "nodes" array')
   125→  }
   126→  const validTypes = new Set(Object.keys(nodeTypes))
   127→  const stamp = Date.now()
   128→  const nodes: Node<ShapeData>[] = (obj.nodes as Record<string, any>[]).map((n, i) => {
   129→    const record = (n ?? {}) as Record<string, any>
   130→    return {
   131→      id: typeof record.id === "string" && record.id ? record.id : `n-import-${stamp}-${i}`,
   132→      type: validTypes.has(record.type) ? record.type : "rectangle",
   133→      position: {
   134→        x: Number.isFinite(Number(record.position?.x)) ? Number(record.position.x) : 0,
   135→        y: Number.isFinite(Number(record.position?.y)) ? Number(record.position.y) : 0,
   136→      },
   137→      data: { ...(record.data ?? {}) },
   138→      style: { ...(record.style ?? {}) },
   139→      selected: false,
   140→      dragging: false,
   141→      resizing: false,
   142→    } as unknown as Node<ShapeData>
   143→  })
   144→  const nodeIds = new Set(nodes.map((n) => n.id))
   145→  const edges: Edge[] = (Array.isArray(obj.edges) ? (obj.edges as Record<string, any>[]) : [])
   146→    .filter(
   147→      (e) =>
   148→        e &&
   149→        typeof e.source === "string" &&
   150→        typeof e.target === "string" &&
   151→        nodeIds.has(e.source) &&
   152→        nodeIds.has(e.target) &&
   153→        e.source !== e.target,
   154→    )
   155→    .map((e, i) => ({
   156→      ...e,
   157→      id: typeof e.id === "string" && e.id ? e.id : `e-import-${stamp}-${i}`,
   158→      type: typeof e.type === "string" && e.type ? e.type : "smart",
   159→      selected: false,
   160→    })) as Edge[]
   161→  const viewport = obj.viewport as { x: number; y: number; zoom: number } | undefined
   162→  return { nodes, edges, viewport }
   163→}
